   219→// ── OTP Login page ────────────────────────────────────────────
   220→function OtpLoginPage({ onBackToSignIn }: { onBackToSignIn: () => void }) {
   221→  const [email, setEmail] = useState('');
   222→  const [otp, setOtp] = useState('');
   223→  const [step, setStep] = useState<'request' | 'verify'>('request');
   224→  const [loading, setLoading] = useState(false);
   225→  const [error, setError] = useState('');
   226→  const [resendCooldown, setResendCooldown] = useState(0);
   227→  const cooldownRef = useRef<ReturnType<typeof setInterval>>();
   228→
   229→  // Cooldown timer for resend
   230→  useEffect(() => {
   231→    if (resendCooldown <= 0) return;
   232→    cooldownRef.current = setInterval(() => {
   233→      setResendCooldown((prev) => {
   234→        if (prev <= 1) {
   235→          clearInterval(cooldownRef.current);
   236→          return 0;
   237→        }
   238→        return prev - 1;
   239→      });
   240→    }, 1000);
   241→    return () => clearInterval(cooldownRef.current);
   242→  }, [resendCooldown]);
   243→
   244→  const handleRequestOtp = useCallback(
   245→    async (e: React.FormEvent) => {
   246→      e.preventDefault();
   247→      if (!email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   248→        setError('Enter a valid email address');
   249→        return;
   250→      }
   251→      setError('');
   252→      setLoading(true);
   253→      try {
   254→        const res = await fetch('/api/auth/otp/request', {
   255→          method: 'POST',
   256→          headers: { 'Content-Type': 'application/json' },
   257→          body: JSON.stringify({ email }),
   258→        });
   259→        const data = await res.json();
   260→        if (res.ok) {
   261→          setStep('verify');
   262→          setResendCooldown(60);
   263→          // Simulation paths permanently disabled — OTP must come via email
   264→          toast.success('If an account exists for this email, you will receive a sign-in code.');
   265→        } else {
   266→          setError(data.error || 'Failed to send OTP');
   267→          toast.error(data.error || 'Failed to send OTP');
   268→        }
   269→      } catch {
   270→        setError('Network error. Please try again.');
   271→      } finally {
   272→        setLoading(false);
   273→      }
   274→    },
   275→    [email]
   276→  );
   277→
   278→  const handleVerifyOtp = useCallback(async () => {
   279→    if (otp.length !== 6) {
   280→      setError('Enter the 6-digit code');
   281→      return;
   282→    }
   283→    setError('');
   284→    setLoading(true);
   285→    try {
   286→      const res = await fetch('/api/auth/otp/verify', {
   287→        method: 'POST',
   288→        headers: { 'Content-Type': 'application/json' },
   289→        credentials: 'include',
   290→        body: JSON.stringify({ email, otp }),
   291→      });
   292→      const data = await res.json();
   293→      if (res.ok) {
   294→        toast.success('Welcome back!');
   295→        // Auth cookies are set by the API, refresh auth state
   296→        window.location.reload();
   297→      } else {
   298→        setError(data.error || 'Invalid OTP');
   299→        toast.error(data.error || 'Verification failed');
   300→      }
   301→    } catch {
   302→      setError('Network error. Please try again.');
   303→    } finally {
   304→      setLoading(false);
   305→    }
   306→  }, [email, otp]);
   307→
   308→  return (
   309→    <AuthLayout
   310→      title="Sign in with OTP"
   311→      description={step === 'request' ? 'Enter your email to receive a login code' : `Enter the 6-digit code sent to ${email}`}
   312→      showBack
   313→      onBack={onBackToSignIn}
   314→    >
   315→      <Card className="glass-card">
   316→        <CardContent className="pt-6 space-y-4">
   317→          <AnimatePresence mode="wait">
   318→            {error && (
   319→              <motion.div
   320→                initial={{ opacity: 0, y: -6, height: 0 }}
   321→                animate={{ opacity: 1, y: 0, height: 'auto' }}
   322→                exit={{ opacity: 0, y: -6, height: 0 }}
   323→                className="rounded-lg border border-destructive/30 bg-destructive/5 p-3 text-sm text-destructive overflow-hidden"
   324→                role="alert"
   325→              >
   326→                <div className="flex items-center gap-2">
   327→                  <AlertTriangle className="h-4 w-4 shrink-0" />
   328→                  {error}
   329→                </div>
   330→              </motion.div>
   331→            )}
   332→          </AnimatePresence>
   333→
   334→          <AnimatePresence mode="wait">
   335→            {step === 'request' ? (
   336→              <motion.form
   337→                key="request"
   338→                onSubmit={handleRequestOtp}
   339→                className="space-y-4"
   340→                noValidate
   341→                initial={{ opacity: 0, x: -20 }}
   342→                animate={{ opacity: 1, x: 0 }}
   343→                exit={{ opacity: 0, x: 20 }}
   344→                transition={{ duration: 0.2 }}
   345→              >
   346→                <div className="space-y-2">
   347→                  <Label htmlFor="otp-email" className="text-xs font-medium">
   348→                    Email
   349→                  </Label>
   350→                  <div className="relative">
   351→                    <Mail className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   352→                    <Input
   353→                      id="otp-email"
   354→                      type="email"
   355→                      placeholder="you@example.com"
   356→                      className="pl-9 h-10 border-0 bg-muted/50 focus-visible:bg-background transition-colors"
   357→                      value={email}
   358→                      onChange={(e) => { setEmail(e.target.value); if (error) setError(''); }}
   359→                      aria-invalid={!!error}
   360→                      autoComplete="email"
   361→                    />
   362→                  </div>
   363→                </div>
   364→                <Button
   365→                  type="submit"
   366→                  disabled={loading}
   367→                  className="w-full h-10 btn-ripple"
   368→                >
   369→                  {loading ? (
   370→                    <>
   371→                      <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   372→                      Sending...
   373→                    </>
   374→                  ) : (
   375→                    <>
   376→                      <Key className="mr-2 h-4 w-4" />
   377→                      Send Login Code
   378→                    </>
   379→                  )}
   380→                </Button>
   381→              </motion.form>
   382→            ) : (
   383→              <motion.div
   384→                key="verify"
   385→                className="space-y-4"
   386→                initial={{ opacity: 0, x: 20 }}
   387→                animate={{ opacity: 1, x: 0 }}
   388→                exit={{ opacity: 0, x: -20 }}
   389→                transition={{ duration: 0.2 }}
   390→              >
   391→                <div className="flex flex-col items-center gap-2">
   392→                  <Label className="text-xs font-medium text-muted-foreground">Verification Code</Label>
   393→                  <InputOTP maxLength={6} value={otp} onChange={setOtp}>
   394→                    <InputOTPGroup>
   395→                      <InputOTPSlot index={0} className="h-12 w-12 text-lg" />
   396→                      <InputOTPSlot index={1} className="h-12 w-12 text-lg" />
   397→                      <InputOTPSlot index={2} className="h-12 w-12 text-lg" />
   398→                    </InputOTPGroup>
   399→                    <InputOTPSeparator />
   400→                    <InputOTPGroup>
   401→                      <InputOTPSlot index={3} className="h-12 w-12 text-lg" />
   402→                      <InputOTPSlot index={4} className="h-12 w-12 text-lg" />
   403→                      <InputOTPSlot index={5} className="h-12 w-12 text-lg" />
   404→                    </InputOTPGroup>
   405→                  </InputOTP>
   406→                </div>
   407→                <Button
   408→                  type="button"
   409→                  disabled={loading || otp.length !== 6}
   410→                  onClick={handleVerifyOtp}
   411→                  className="w-full h-10 btn-ripple"
   412→                >
   413→                  {loading ? (
   414→                    <>
   415→                      <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   416→                      Verifying...
   417→                    </>
   418→                  ) : (
   419→                    'Verify & Sign In'
   420→                  )}
   421→                </Button>
   422→                {resendCooldown > 0 ? (
   423→                  <p className="text-xs text-center text-muted-foreground">
   424→                    Resend in <span className="font-mono text-primary">{resendCooldown}s</span>
   425→                  </p>
   426→                ) : (
   427→                  <button
   428→                    type="button"
   429→                    onClick={() => { handleRequestOtp({ preventDefault: () => {} } as React.FormEvent); }}
   430→                    className="text-xs text-center text-primary hover:text-primary/80 font-medium transition-colors w-full"
   431→                  >
   432→                    Resend code
   433→                  </button>
   434→                )}
   435→              </motion.div>
   436→            )}
   437→          </AnimatePresence>
   438→        </CardContent>
   439→      </Card>
   440→    </AuthLayout>
   441→  );
   442→}

   556→                        if (error) setError('');
   557→                      }}
   558→                      aria-invalid={!!error}
   559→                      autoComplete="email"
   560→                    />
   561→                  </div>
   562→                </div>
   563→                <Button
   564→                  type="submit"
   565→                  disabled={loading}
   566→                  className="w-full h-10 btn-ripple"
   567→                >
   568→                  {loading ? (
   569→                    <>
   570→                      <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   571→                      Sending...
   572→                    </>
   573→                  ) : (
   574→                    <>
   575→                      <Mail className="mr-2 h-4 w-4" />
   576→                      Send Magic Link
   577→                    </>
   578→                  )}
   579→                </Button>
   580→              </motion.form>
   581→            )}
   582→          </AnimatePresence>
   583→        </CardContent>
   584→      </Card>
   585→    </AuthLayout>
   586→  );
   587→}

   631→// Main AuthGate component
   632→// ══════════════════════════════════════════════════════════════
   633→export default function AuthGate() {
   634→  const { isAuthenticated, isLoading, mfaRequired, setUser, setLoading } = useAuthStore();
   635→  const { signIn } = useAuth();
   636→
   637→  // ── Auth page navigation state ───────────────────────────────
   638→  const [authPage, setAuthPage] = useState<AuthPage>('signin');
   639→  const [flowEmail, setFlowEmail] = useState('');
   640→  const [initialized, setInitialized] = useState(false);
   641→
   642→  // ── Subscription banner state (authenticated only) ──────────
   643→  const [upgradeModalOpen, setUpgradeModalOpen] = useState(false);
   644→  const [creditWarningDismissed, setCreditWarningDismissed] = useState(false);
   645→
   646→  // ── Auto token refresh when authenticated ────────────────────
   647→  useTokenRefresh();
   648→
   649→  // ── Initialize: check auth state on mount ────────────────────
   650→  useEffect(() => {
   651→    const init = async () => {
   652→      // Check for OAuth error redirect
   653→      const params = new URLSearchParams(window.location.search);
   654→      const authError = params.get('auth_error');
   655→      if (authError === 'google_failed') {
   656→        toast.error('Google login failed. Please try again or use email/password.');
   657→        window.history.replaceState({}, '', window.location.pathname);
   658→      }
   659→
   660→      // Check for payment success/cancel redirect from Stripe
   661→      const paymentStatus = params.get('payment');
   662→      if (paymentStatus === 'success') {
   663→        const plan = params.get('plan');
   664→        const orderId = params.get('orderId');
   665→        if (orderId) {
   666→          // Confirm payment with backend (dev mode only — production uses webhooks)
   667→          try {
   668→            await fetch('/api/payments/confirm', {
   669→              method: 'POST',
   670→              headers: { 'Content-Type': 'application/json' },
   671→              credentials: 'include',
   672→              body: JSON.stringify({ orderId, providerPaymentId: `stripe_checkout_${Date.now()}` }),
   673→            });
   674→            toast.success(`Successfully upgraded to ${plan || 'selected'} plan!`);
   675→          } catch {
   676→            toast.info('Payment received. Your plan will be updated shortly.');
   677→          }
   678→        } else {
   679→          toast.success('Payment successful! Your plan will be updated shortly.');
   680→        }
   681→        window.history.replaceState({}, '', window.location.pathname);
   682→      } else if (paymentStatus === 'cancelled') {
   683→        toast.info('Payment was cancelled.');
   684→        window.history.replaceState({}, '', window.location.pathname);
   685→      }
   686→
   687→      try {
   688→        const res = await fetch('/api/auth/me', { credentials: 'include' });
   689→        if (res.ok) {
   690→          const data = await res.json();
   691→          setUser(data.user);
   692→        } else {
   693→          setUser(null);
   694→        }
   695→      } catch {
   696→        setUser(null);
   697→      } finally {
   698→        setInitialized(true);
   699→      }
   700→    };
   701→    init();
   702→  }, [setUser]);
   703→
   704→  // ── Auto-switch to MFA page when mfaRequired becomes true ───
   705→  useEffect(() => {
   706→    if (mfaRequired && authPage !== 'mfa-verify') {
   707→      setAuthPage('mfa-verify');
   708→    }
   709→  }, [mfaRequired, authPage]);
   710→
   711→  // ── Navigation handlers ──────────────────────────────────────
   712→  const navigateToSignIn = useCallback(() => {
   713→    setAuthPage('signin');
   714→    setFlowEmail('');
   715→  }, []);
   716→
   717→  const navigateToSignUp = useCallback(() => {
   718→    setAuthPage('signup');
   719→  }, []);
   720→
   721→  const navigateToForgotPassword = useCallback(() => {
   722→    setAuthPage('forgot-password');
   723→  }, []);
   724→
   725→  const navigateToVerifyEmail = useCallback((email: string) => {
   726→    setFlowEmail(email);
   727→    setAuthPage('verify-email');
   728→  }, []);
   729→
   730→  const [resetDevOtp, setResetDevOtp] = useState<string | undefined>();
   731→
   732→  const navigateToResetPassword = useCallback((email: string, devOtp?: string) => {
   733→    setFlowEmail(email);
   734→    setResetDevOtp(devOtp);
   735→    setAuthPage('reset-password');
   736→  }, []);
   737→
   738→  const navigateToMagicLink = useCallback(() => {
   739→    setAuthPage('magic-link');
   740→  }, []);
   741→
   742→  const navigateToOtpLogin = useCallback(() => {
   743→    setAuthPage('otp-login');
   744→  }, []);
   745→
   746→  // ── Sign in success handler ──────────────────────────────────
   747→  const handleSignInSuccess = useCallback(() => {
   748→    // The signIn in useAuth already sets the user in the store.
   749→    // If MFA was required, the mfaRequired effect above will switch to the MFA page.
   750→    // If no MFA, the user is set and isAuthenticated becomes true, showing the dashboard.
   751→  }, []);
   752→
   753→  // ── Email verified handler: auto-sign-in after verification ──
   754→  const handleEmailVerified = useCallback(async () => {
   755→    if (flowEmail) {
   756→      toast.info('Signing you in...');
   757→      // After email verification, attempt auto sign-in
   758→      // The user needs to have set a password during signup, so we prompt them to sign in
   759→      setAuthPage('signin');
   760→      toast.success('Email verified! Please sign in with your credentials.');
   761→    } else {
   762→      navigateToSignIn();
   763→    }
   764→  }, [flowEmail, navigateToSignIn]);
   765→
   766→  // ── Password reset success handler ───────────────────────────
   767→  const handleResetPasswordSuccess = useCallback(() => {
   768→    toast.success('Password reset! Please sign in with your new password.');
   769→    navigateToSignIn();
   770→  }, [navigateToSignIn]);
   771→
   772→  // ── MFA verification success handler ─────────────────────────
   773→  const handleMfaSuccess = useCallback(() => {
   774→    // The verifyMfa in useAuth already sets the user in the store
   775→    // isAuthenticated will become true and the dashboard will show
   776→    toast.success('Welcome back!');
   777→  }, []);
   778→
   779→  // ── Legal pages dialog (available everywhere) ────────────────
   780→  const legalDialog = <LegalPages />;
   781→
   782→  // ── Loading state while checking auth ────────────────────────
   783→  if (!initialized || isLoading) {
   784→    return (
   785→      <>
   786→        {legalDialog}
   787→        <LoadingScreen />
   788→      </>
   789→    );
   790→  }
   791→
   792→  // ── Authenticated: show dashboard ────────────────────────────
   793→  if (isAuthenticated && !mfaRequired) {
   794→    return (
   795→      <>
   796→        {legalDialog}
   797→        <AuthenticatedWrapper
   798→          upgradeModalOpen={upgradeModalOpen}
   799→          setUpgradeModalOpen={setUpgradeModalOpen}
   800→          creditWarningDismissed={creditWarningDismissed}
   801→          setCreditWarningDismissed={setCreditWarningDismissed}
   802→        />
   803→      </>
   804→    );
   805→  }
   806→
   807→  // ── Unauthenticated: show auth pages ─────────────────────────
   808→  switch (authPage) {
   809→    case 'signin':
   810→      return (
   811→        <>
   812→          {legalDialog}
   813→          <SignInPage
   814→            onSignUpClick={navigateToSignUp}
   815→            onForgotPasswordClick={navigateToForgotPassword}
   816→            onSuccess={handleSignInSuccess}
   817→            onMagicLinkClick={navigateToMagicLink}
   818→            onOtpLoginClick={navigateToOtpLogin}
   819→          />
   820→        </>
   821→      );
   822→
   823→    case 'signup':
   824→      return (
   825→        <>
   826→          {legalDialog}
   827→          <SignUpPage
   828→            onSignInClick={navigateToSignIn}
   829→            onVerifyEmail={navigateToVerifyEmail}
   830→          />
   831→        </>
   832→      );
   833→
   834→    case 'verify-email':
   835→      return (
   836→        <>
   837→          {legalDialog}
   838→          <VerifyEmailPage
   839→            email={flowEmail}
   840→            onVerified={handleEmailVerified}
   841→            onBackToSignIn={navigateToSignIn}
   842→          />
   843→        </>
   844→      );
   845→
   846→    case 'forgot-password':
   847→      return (
   848→        <>
   849→          {legalDialog}
   850→          <ForgotPasswordPage
   851→            onBackToSignIn={navigateToSignIn}
   852→            onOtpSent={navigateToResetPassword}
   853→          />
   854→        </>
   855→      );
   856→
   857→    case 'reset-password':
   858→      return (
   859→        <>
   860→          {legalDialog}
   861→          <ResetPasswordPage
   862→            email={flowEmail}
   863→            onSuccess={handleResetPasswordSuccess}
   864→            initialOtp={resetDevOtp}
   865→          />
   866→        </>
   867→      );
   868→
   869→    case 'mfa-verify':
   870→      return (
   871→        <>
   872→          {legalDialog}
   873→          <MfaVerificationPage
   874→            onSuccess={handleMfaSuccess}
   875→            onBack={navigateToSignIn}
   876→          />
   877→        </>
   878→      );
   879→
   880→    case 'otp-login':
   881→      return (
   882→        <>
   883→          {legalDialog}
   884→          <OtpLoginPage onBackToSignIn={navigateToSignIn} />
   885→        </>
   886→      );
   887→
   888→    case 'magic-link':
   889→      return (
   890→        <>
   891→          {legalDialog}
   892→          <MagicLinkPage onBackToSignIn={navigateToSignIn} />
   893→        </>
   894→      );
   895→
   896→    default:
   897→      return (
   898→        <>
   899→          {legalDialog}
   900→          <SignInPage
   901→            onSignUpClick={navigateToSignUp}
   902→            onForgotPasswordClick={navigateToForgotPassword}
   903→            onSuccess={handleSignInSuccess}
   904→            onMagicLinkClick={navigateToMagicLink}
   905→            onOtpLoginClick={navigateToOtpLogin}
   906→          />
   907→        </>
   908→      );
