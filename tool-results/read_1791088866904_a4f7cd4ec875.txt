    54→    54→/** Robust AI JSON extraction (same hardened pattern as company-researcher). */
    55→    55→export function parseAIJson<T>(content: string, fallback: T): T {
    56→    56→  let cleaned = content.trim();
    57→    57→  const codeBlockMatch = cleaned.match(/^```(?:json)?\s*\n?([\s\S]*?)\n?```\s*$/);
    58→    58→  if (codeBlockMatch) cleaned = codeBlockMatch[1].trim();
    59→    59→  else if (cleaned.startsWith('```')) {
    60→    60→    cleaned = cleaned.replace(/^```(?:json)?\s*\n?/, '').replace(/\n?```\s*$/, '');
    61→    61→  }
    62→    62→  const jsonMatch = cleaned.match(/\{[\s\S]*\}/);
    63→    63→  if (!jsonMatch) {
    64→    64→    console.error('[ProspectPipeline] No JSON object found in AI response');
    65→    65→    return fallback;
    66→    66→  }
    67→    67→  try {
    68→    68→    return JSON.parse(jsonMatch[0]) as T;
    69→    69→  } catch (err) {
    70→    70→    console.error('[ProspectPipeline] JSON.parse failed:', err);
    71→    71→    return fallback;
    72→    72→  }
    73→    73→}

    75→    75→async function aiJson(system: string, user: string, userId: string, action: string, maxTokens = 1600) {
    76→    76→  const result = await executeAICompletion(
    77→    77→    {

    80→    80→        { role: 'user', content: user },
    81→    81→      ],
    82→    82→      config: { provider: 'z-ai', temperature: 0.35, maxTokens },
    83→    83→    },
    84→    84→    userId,
    85→    85→    action
    86→    86→  );
    87→    87→  if (!result.success) {
    88→    88→    throw new Error(result.error || 'AI provider failed');
    89→    89→  }
    90→    90→  return result;
    91→    91→}

    94→    94→async function webSearchSnippets(lead: LeadContext): Promise<{ used: boolean; text: string }> {
    95→    95→  try {
    96→    96→    const zai = await ZAI.create();
    97→    97→    const queries = [
    98→    98→      `${lead.businessName} ${lead.city || ''} reviews`.trim(),
    99→    99→      `${lead.businessName} ${lead.niche || 'company'} services team`,
   100→   100→    ];
   101→   101→    const chunks: string[] = [];
   102→   102→    for (const query of queries) {
   103→   103→      try {
   104→   104→        const results = await zai.functions.invoke('web_search', { query, num: 5 });
   105→   105→        if (Array.isArray(results) && results.length > 0) {
   106→   106→          chunks.push(
   107→   107→            results
   108→   108→              .map((r) => {
   109→   109→                const name = typeof r?.name === 'string' ? r.name : '';
   110→   110→                const snippet = typeof r?.snippet === 'string' ? r.snippet : '';
   111→   111→                const link = typeof r?.url === 'string' ? r.url : '';
   112→   112→                return `- ${name} ${snippet} ${link}`.trim();
   113→   113→              })
   114→   114→              .filter((s: string) => s.length > 3)
   115→   115→              .join('\n')
   116→   116→          );
   117→   117→        }
   118→   118→      } catch (qErr) {
   119→   119→        console.error('[ProspectPipeline] web_search query failed:', qErr);
   120→   120→      }
   121→   121→    }
   122→   122→    const text = chunks.join('\n').slice(0, 6000);
   123→   123→    return { used: text.length > 0, text };
   124→   124→  } catch (err) {
   125→   125→    console.error('[ProspectPipeline] web_search unavailable:', err);
   126→   126→    return { used: false, text: '' };
   127→   127→  }
   128→   128→}

   401→   401→      severity: 'minor',
   402→   402→      businessImpact: 'Missed trust-building and re-marketing opportunities',
   403→   403→    });
   404→   404→  }
   405→   405→
   406→   406→  return { gaps, strengths };

   453→   453→    const parsed = parseAIJson<{ gaps?: Partial<PipelineGap>[] }>(result.content, {});
   454→   454→    aiGaps = (Array.isArray(parsed.gaps) ? parsed.gaps : [])
   455→   455→      .slice(0, 6)
   456→   456→      .map((g) => ({
   457→   457→        category:
   458→   458→          typeof g?.category === 'string' && validCats.has(g.category)
   459→   459→            ? (g.category as PipelineGap['category'])
   460→   460→            : 'weak_area' as PipelineGap['category'],
   461→   461→        gap: typeof g?.gap === 'string' ? g.gap.trim().slice(0, 200) : '',
   462→   462→        evidence: typeof g?.evidence === 'string' ? g.evidence.trim().slice(0, 300) : '',
   463→   463→        severity:
   464→   464→          g?.severity === 'critical' || g?.severity === 'minor'
   465→   465→            ? (g.severity as PipelineGap['severity'])
   466→   466→            : ('moderate' as PipelineGap['severity']),
   467→   467→        businessImpact:
   468→   468→          typeof g?.businessImpact === 'string' ? g.businessImpact.trim().slice(0, 300) : '',
   469→   469→      }))
   470→   470→      .filter((g) => g.gap.length > 0);
   471→   471→  } catch (err) {
   472→   472→    console.error('[ProspectPipeline] AI gap pass failed (continuing with rule gaps):', err);
   473→   473→  }
   474→   474→
   475→   475→  // Merge + dedupe (case-insensitive containment check)
   476→   476→  const merged: PipelineGap[] = [...ruleResult.gaps];
   477→   477→  for (const aiGap of aiGaps) {
   478→   478→    const lower = aiGap.gap.toLowerCase();
   479→   479→    const duplicate = merged.some((m) => {
   480→   480→      const ml = m.gap.toLowerCase();
   481→   481→      return ml === lower || ml.includes(lower) || lower.includes(ml);
   482→   482→    });
   483→   483→    if (!duplicate) merged.push(aiGap);
   484→   484→  }
   485→   485→
   486→   486→  const severityRank = { critical: 0, moderate: 1, minor: 2 } as const;
   487→   487→  merged.sort((a, b) => severityRank[a.severity] - severityRank[b.severity]);
   488→   488→
   489→   489→  return {
   490→   490→    overallDigitalHealthScore: websiteScore?.overallScore ?? 0,
   491→   491→    gaps: merged,
   492→   492→    strengths: ruleResult.strengths,
   493→   493→  };
   494→   494→}

   498→   498→/** Deterministic keyword-overlap fallback matcher (used only if AI fails). */
   499→   499→function keywordMatchOffers(
   500→   500→  offers: OfferService[],
   501→   501→  gaps: PipelineGap[]
   502→   502→): OfferMatch[] {
   503→   503→  const stop = new Set(['and', 'the', 'for', 'with', 'a', 'an', 'of', 'in', 'to', 'no', 'not']);
   504→   504→  const tokenize = (s: string) =>
   505→   505→    s
   506→   506→      .toLowerCase()
   507→   507→      .replace(/[^a-z0-9\s]/g, ' ')
   508→   508→      .split(/\s+/)
   509→   509→      .filter((t) => t.length > 2 && !stop.has(t));
   510→   510→
   511→   511→  return offers
   512→   512→    .map((offer) => {
   513→   513→      const offerTokens = new Set(tokenize(`${offer.label} ${offer.category} ${offer.description}`));
   514→   514→      const matchedGaps: string[] = [];
   515→   515→      for (const gap of gaps) {
   516→   516→        const gapTokens = tokenize(`${gap.gap} ${gap.category}`);
   517→   517→        const overlap = gapTokens.filter((t) => offerTokens.has(t));
   518→   518→        if (overlap.length >= 2 || (overlap.length === 1 && gap.severity === 'critical')) {
   519→   519→          matchedGaps.push(gap.gap);
   520→   520→        }
   521→   521→      }
   522→   522→      if (matchedGaps.length === 0) return null;
   523→   523→      const strength: OfferMatch['matchStrength'] =
   524→   524→        matchedGaps.length >= 3 ? 'strong' : matchedGaps.length === 2 ? 'partial' : 'weak';
   525→   525→      return {
   526→   526→        offerLabel: offer.label,
   527→   527→        matchedGaps,
   528→   528→        matchStrength: strength,
   529→   529→        rationale: `Matches ${matchedGaps.length} detected gap(s) via service keywords`,
   530→   530→        angle: `Position ${offer.label} as the direct fix for: ${matchedGaps[0]}`,
   531→   531→      } satisfies OfferMatch;
   532→   532→    })
   533→   533→    .filter((m): m is OfferMatch => m !== null)
   534→   534→    .slice(0, 4);
   535→   535→}
