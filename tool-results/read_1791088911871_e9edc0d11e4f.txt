    36→/**
    37→ * Execute a workflow action based on its type and config.
    38→ * Uses existing services where possible.
    39→ */
    40→export async function executeAction(
    41→  actionType: string,
    42→  config: Record<string, unknown>,
    43→  context: ActionContext
    44→): Promise<ActionResult> {
    45→  try {
    46→    switch (actionType) {
    47→      case 'send_email':
    48→        return await executeSendEmail(config, context);
    49→      case 'create_gmail_draft':
    50→        return await executeCreateGmailDraft(config, context);
    51→      case 'send_telegram':
    52→        return await executeSendTelegram(config, context);
    53→      case 'send_whatsapp':
    54→        return await executeSendWhatsapp(config, context);
    55→      case 'ai_analysis':
    56→        return await executeAiAnalysis(config, context);
    57→      case 'ai_outreach':
    58→        return await executeAiOutreach(config, context);
    59→      case 'move_lead_stage':
    60→        return await executeMoveLeadStage(config, context);
    61→      case 'update_tags':
    62→        return await executeUpdateTags(config, context);
    63→      case 'create_notification':
    64→        return await executeCreateNotification(config, context);
    65→      case 'wait_delay':
    66→        return await executeWaitDelay(config, context);
    67→      case 'conditional_branch':
    68→        return await executeConditionalBranch(config, context);
    69→      case 'webhook_call':
    70→        return await executeWebhookCall(config, context);
    71→      case 'export_data':
    72→        return await executeExportData(config, context);
    73→      case 'score_lead':
    74→        return await executeScoreLead(config, context);
    75→      case 'add_note':
    76→        return await executeAddNote(config, context);
    77→      case 'notify_low_credits':
    78→        return await executeNotifyLowCredits(config, context);
    79→      case 'notify_trial_ending':
    80→        return await executeNotifyTrialEnding(config, context);
    81→      default:
    82→        return { success: false, error: `Unknown action type: ${actionType}` };
    83→    }
    84→  } catch (error) {
    85→    return {
    86→      success: false,
    87→      error: error instanceof Error ? error.message : 'Action execution failed',
    88→    };
    89→  }
    90→}

    94→async function executeSendEmail(
    95→  config: Record<string, unknown>,
    96→  context: ActionContext
    97→): Promise<ActionResult> {
    98→  const { to, subject, body } = config;
    99→
   100→  if (!to || !subject || !body) {
   101→    return { success: false, error: 'Email requires to, subject, and body' };
   102→  }
   103→
   104→  // Resolve template variables in subject/body
   105→  const resolvedSubject = resolveTemplate(String(subject), context);
   106→  const resolvedBody = resolveTemplate(String(body), context);
   107→
   108→  const result = await sendEmail({
   109→    to: String(to),
   110→    subject: resolvedSubject,
   111→    html: resolvedBody,
   112→    text: resolvedBody,
   113→  });
   114→
   115→  return {
   116→    success: result.sent,
   117→    output: { messageId: result.messageId, devMode: result.devMode },
   118→    error: result.error,
   119→    creditsUsed: 2,
   120→  };
   121→}

   280→      success: true,
   281→      output: {
   282→        analysis: response.text || response.content || '',
   283→        leadId: context.leadId,
   284→        model: model || 'default',
   285→      },

   296→async function executeAiOutreach(
   297→  config: Record<string, unknown>,
   298→  context: ActionContext
   299→): Promise<ActionResult> {
   300→  const { channel, style, tone } = config;
   301→
   302→  if (!context.leadId) {
   303→    return { success: false, error: 'AI outreach requires a lead context' };
   304→  }
   305→
   306→  try {
   307→    const chat = await createZaiChat();
   308→
   309→    const resolution = await resolveLeadForExecution(context.userId, context.leadId);
   310→    if (!resolution.ok) {
   311→      return { success: false, error: 'Lead not found for AI outreach' };
   312→    }
   313→    const lead = resolution.lead;
   314→
   315→    const outreachChannel = String(channel || 'email');
   316→    const outreachStyle = style || 'professional';
   317→    const outreachTone = tone || 'friendly';
   318→
   319→    // SENDER IDENTITY: the workflow acts on behalf of the user who owns it.
   320→    // Real profile goes into the prompt; placeholder tokens the model still
   321→    // emits are replaced before the draft is stored.
   322→    const senderProfile = await loadSenderProfile(context.userId);
   323→    const senderBlock = buildSenderSignatureBlock(senderProfile);
   324→
   325→    const prompt = `Generate a ${outreachStyle} ${outreachChannel} outreach message for:
   326→      Business: ${lead.businessName}
   327→      Owner: ${lead.ownerName || 'Unknown'}
   328→      Niche: ${lead.niche || 'Unknown'}
   329→      City: ${lead.city || 'Unknown'}
   330→      Tone: ${outreachTone}
   331→      
   332→      Create a compelling, personalized message that addresses their potential needs.
   333→
   334→${senderBlock}`;
   335→
   336→    const response = await chat.sendMessage(prompt);
   337→    const generatedMessage = applySenderSignature(
   338→      response.text || response.content || '',
   339→      senderProfile
   340→    );
   341→
   342→    // Create outreach message record
   343→    const message = await db.outreachMessage.create({
   344→      data: {
   345→        leadId: context.leadId,
   346→        userId: context.userId,
   347→        channel: outreachChannel,
   348→        direction: 'outbound',
   349→        subject: outreachChannel === 'email' ? `Re: ${lead.businessName}` : null,
   350→        content: generatedMessage,
   351→        status: 'draft',
   352→        generatedByAI: true,
   353→      },
   354→    });
   355→
   356→    return {
   357→      success: true,
   358→      output: {
   359→        messageId: message.id,
   360→        channel: outreachChannel,
   361→        generatedMessage,
   362→      },
   363→      creditsUsed: 5,
   364→    };
   365→  } catch (error) {
   366→    return {
   367→      success: false,
   368→      error: `AI outreach failed: ${error instanceof Error ? error.message : 'Unknown error'}`,
   369→    };
   370→  }
   371→}

   408→async function executeUpdateTags(
   409→  config: Record<string, unknown>,
   410→  context: ActionContext
   411→): Promise<ActionResult> {
   412→  const { tags, action } = config;
   413→
   414→  if (!context.leadId) {
   415→    return { success: false, error: 'Update tags requires a lead context' };
   416→  }
   417→
   418→  if (!tags || !Array.isArray(tags)) {
   419→    return { success: false, error: 'Tags array is required' };
   420→  }
   421→
   422→  const tagsResolution = await resolveLeadForExecution(context.userId, context.leadId, {
   423→    select: { tags: true },
   424→  });
   425→
   426→  if (!tagsResolution.ok) {
   427→    return { success: false, error: 'Lead not found' };
   428→  }
   429→  const lead = tagsResolution.lead;
   430→
   431→  let currentTags: string[] = [];
   432→  try {
   433→    currentTags = JSON.parse(lead.tags || '[]');
   434→  } catch {
   435→    currentTags = [];
   436→  }
   437→
   438→  const tagAction = action || 'add';
   439→  let newTags: string[];
   440→
   441→  if (tagAction === 'add') {
   442→    newTags = [...new Set([...currentTags, ...tags.map(String)])];
   443→  } else if (tagAction === 'remove') {
   444→    newTags = currentTags.filter(t => !tags.map(String).includes(t));
   445→  } else {
   446→    newTags = tags.map(String);
   447→  }
   448→
   449→  await db.lead.update({
   450→    where: { id: context.leadId },
   451→    data: { tags: JSON.stringify(newTags) },
   452→  });
   453→
   454→  return {
   455→    success: true,
   456→    output: { previousTags: currentTags, newTags, action: tagAction },
   457→    creditsUsed: 1,
   458→  };
   459→}

   489→async function executeWaitDelay(
   490→  config: Record<string, unknown>,
   491→  _context: ActionContext
   492→): Promise<ActionResult> {
   493→  const { duration, unit } = config;
   494→
   495→  if (!duration) {
   496→    return { success: false, error: 'Delay requires a duration' };
   497→  }
   498→
   499→  const durationNum = Number(duration);
   500→  const delayUnit = String(unit || 'minutes');
   501→
   502→  // Convert to milliseconds
   503→  let delayMs: number;
   504→  switch (delayUnit) {
   505→    case 'seconds': delayMs = durationNum * 1000; break;
   506→    case 'minutes': delayMs = durationNum * 60 * 1000; break;
   507→    case 'hours': delayMs = durationNum * 60 * 60 * 1000; break;
   508→    case 'days': delayMs = durationNum * 24 * 60 * 60 * 1000; break;
   509→    default: delayMs = durationNum * 60 * 1000; break;
   510→  }
   511→
   512→  return {
   513→    success: true,
   514→    output: { delayMs, duration: durationNum, unit: delayUnit },
   515→    creditsUsed: 0,
   516→  };
   517→}

   519→async function executeConditionalBranch(
   520→  config: Record<string, unknown>,
   521→  context: ActionContext
   522→): Promise<ActionResult> {
   523→  const { field, operator, value, trueBranch, falseBranch } = config;
   524→
   525→  if (!field || !operator) {
   526→    return { success: false, error: 'Conditional branch requires field and operator' };
   527→  }
   528→
   529→  // Get the field value from context
   530→  let fieldValue: unknown;
   531→  if (field === 'leadStage' && context.leadId) {
   532→    const lead = await db.lead.findUnique({
   533→      where: { id: context.leadId },
   534→      select: { stage: true },
   535→    });
   536→    fieldValue = lead?.stage;
   537→  } else if (field === 'leadScore' && context.leadId) {
   538→    const lead = await db.lead.findUnique({
   539→      where: { id: context.leadId },
   540→      select: { conversionScore: true },
   541→    });
   542→    fieldValue = lead?.conversionScore;
   543→  } else if (field === 'triggerData') {
   544→    fieldValue = context.triggerData?.[String(field)];
   545→  } else if (context.previousOutputs) {
   546→    fieldValue = context.previousOutputs[String(field)];
   547→  } else {
   548→    fieldValue = context.triggerData?.[String(field)];
   549→  }
   550→
   551→  // Evaluate condition
   552→  let conditionMet = false;
   553→  const compareValue = value;
   554→
   555→  switch (operator) {
   556→    case 'eq': case '==': conditionMet = fieldValue == compareValue; break;
   557→    case 'neq': case '!=': conditionMet = fieldValue != compareValue; break;
   558→    case 'gt': case '>': conditionMet = Number(fieldValue) > Number(compareValue); break;
   559→    case 'gte': case '>=': conditionMet = Number(fieldValue) >= Number(compareValue); break;
   560→    case 'lt': case '<': conditionMet = Number(fieldValue) < Number(compareValue); break;
   561→    case 'lte': case '<=': conditionMet = Number(fieldValue) <= Number(compareValue); break;
   562→    case 'contains': conditionMet = String(fieldValue).includes(String(compareValue)); break;
   563→    case 'not_contains': conditionMet = !String(fieldValue).includes(String(compareValue)); break;
   564→    case 'in': conditionMet = String(compareValue).split(',').map(s => s.trim()).includes(String(fieldValue)); break;
   565→    case 'not_in': conditionMet = !String(compareValue).split(',').map(s => s.trim()).includes(String(fieldValue)); break;
   566→    default: conditionMet = false;
   567→  }
   568→
   569→  return {
   570→    success: true,
   571→    output: {
   572→      conditionMet,
   573→      branch: conditionMet ? (trueBranch || 'true') : (falseBranch || 'false'),
   574→      fieldValue,
   575→      operator,
   576→      compareValue,
   577→    },
   578→    creditsUsed: 0,
   579→  };
   580→}

   582→async function executeWebhookCall(
   583→  config: Record<string, unknown>,
   584→  context: ActionContext
   585→): Promise<ActionResult> {
   586→  const { url, method, headers, body } = config;
   587→
   588→  if (!url) {
   589→    return { success: false, error: 'Webhook requires a URL' };
   590→  }
   591→
   592→  try {
   593→    const response = await fetch(String(url), {
   594→      method: String(method || 'POST'),
   595→      headers: {
   596→        'Content-Type': 'application/json',
   597→        ...(headers as Record<string, string> || {}),
   598→      },
   599→      body: body ? JSON.stringify({
   600→        ...((typeof body === 'object' ? body : {}) as Record<string, unknown>),
   601→        workflowId: context.workflowId,
   602→        executionId: context.executionId,
   603→        leadId: context.leadId,
   604→        triggerData: context.triggerData,
   605→      }) : JSON.stringify({
   606→        workflowId: context.workflowId,
   607→        executionId: context.executionId,
   608→        leadId: context.leadId,
   609→        triggerData: context.triggerData,
   610→      }),
   611→    });
   612→
   613→    let responseBody: unknown;
   614→    try {
   615→      responseBody = await response.json();
   616→    } catch {
   617→      responseBody = await response.text();
   618→    }
   619→
   620→    return {
   621→      success: response.ok,
   622→      output: {
   623→        statusCode: response.status,
   624→        body: responseBody,
   625→      },
   626→      error: response.ok ? undefined : `Webhook returned ${response.status}`,
   627→      creditsUsed: 1,
   628→    };
   629→  } catch (error) {
   630→    return {
   631→      success: false,
   632→      error: `Webhook call failed: ${error instanceof Error ? error.message : 'Unknown error'}`,
   633→    };
   634→  }
   635→}

   667→async function executeScoreLead(
   668→  config: Record<string, unknown>,
   669→  context: ActionContext
   670→): Promise<ActionResult> {
   671→  if (!context.leadId) {
   672→    return { success: false, error: 'Score lead requires a lead context' };
   673→  }
   674→
   675→  try {
   676→    const chat = await createZaiChat();
   677→
   678→    const resolution = await resolveLeadForExecution(context.userId, context.leadId);
   679→    if (!resolution.ok) {
   680→      return { success: false, error: 'Lead not found' };
   681→    }
   682→    const lead = resolution.lead;
   683→
   684→    const prompt = `Score this lead on a scale of 0-100 for sales conversion potential:
   685→      Business: ${lead.businessName}
   686→      Niche: ${lead.niche || 'Unknown'}
   687→      Current stage: ${lead.stage}
   688→      Reply score: ${lead.replyScore}
   689→      Conversion score: ${lead.conversionScore}
   690→      
   691→      Return ONLY a number between 0-100.`;
   692→
   693→    const response = await chat.sendMessage(prompt);
   694→    const scoreText = (response.text || response.content || '0').trim();
   695→    const score = Math.min(100, Math.max(0, parseInt(scoreText.replace(/[^0-9]/g, ''), 10) || 0));
   696→
   697→    // Update lead scores
   698→    await db.lead.update({
   699→      where: { id: context.leadId },
   700→      data: { conversionScore: score },
   701→    });
   702→
   703→    // Create score record
   704→    await db.leadScore.create({
   705→      data: {
   706→        leadId: context.leadId,
   707→        scoreType: 'conversion',
   708→        score,
   709→        explanation: scoreText,
   710→      },
   711→    });
   712→
   713→    return {
   714→      success: true,
   715→      output: { leadId: context.leadId, score, explanation: scoreText },
   716→      creditsUsed: 5,
   717→    };
   718→  } catch (error) {
   719→    return {
   720→      success: false,
   721→      error: `Score lead failed: ${error instanceof Error ? error.message : 'Unknown error'}`,
   722→    };
   723→  }
   724→}

   858→// ===== TEMPLATE VARIABLE RESOLVER =====
   859→
   860→function resolveTemplate(template: string, context: ActionContext): string {
   861→  let resolved = template;
   862→
   863→  // Replace common variables
   864→  const replacements: Record<string, string> = {
   865→    '{{leadId}}': context.leadId || '',
   866→    '{{workflowId}}': context.workflowId,
   867→    '{{executionId}}': context.executionId,
   868→    '{{userId}}': context.userId,
   869→  };
   870→
   871→  // Add trigger data variables
   872→  if (context.triggerData) {
   873→    for (const [key, value] of Object.entries(context.triggerData)) {
   874→      replacements[`{{trigger.${key}}}`] = String(value ?? '');
   875→    }
   876→  }
   877→
   878→  // Add previous output variables
   879→  if (context.previousOutputs) {
   880→    for (const [key, value] of Object.entries(context.previousOutputs)) {
   881→      replacements[`{{output.${key}}}`] = String(value ?? '');
   882→    }
   883→  }
   884→
   885→  for (const [placeholder, value] of Object.entries(replacements)) {
   886→    resolved = resolved.replace(new RegExp(escapeRegExp(placeholder), 'g'), value);
   887→  }
   888→
   889→  return resolved;
   890→}
   891→
   892→function escapeRegExp(string: string): string {
   893→  return string.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
   894→}

   896→// ── Local adapter ─────────────────────────────────────────────────
   897→// The z-ai-web-dev-sdk package does not export `createChat`; emulate the
   898→// chat.sendMessage() interface used above via ZAI.create().
   899→
   900→async function createZaiChat(): Promise<{
   901→  sendMessage: (prompt: string) => Promise<{ text?: string; content?: string }>;
   902→}> {
   903→  const { default: ZAI } = await import('z-ai-web-dev-sdk');
   904→  const zai = await ZAI.create();
   905→  return {
   906→    sendMessage: async (prompt: string) => {
   907→      const completion = (await zai.chat.completions.create({
   908→        messages: [{ role: 'user', content: prompt }],
   909→      })) as { choices?: Array<{ message?: { content?: string } }> };
   910→      const text = completion.choices?.[0]?.message?.content ?? '';
   911→      return { text, content: text };
   912→    },
   913→  };
   914→}
