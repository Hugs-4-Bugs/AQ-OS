    48→/** Password strength calculator */
    49→function getPasswordStrength(password: string): {
    50→  score: number; // 0–4
    51→  label: string;
    52→  color: string;
    53→} {
    54→  if (!password) return { score: 0, label: '', color: '' };
    55→
    56→  let score = 0;
    57→  if (password.length >= 8) score++;
    58→  if (password.length >= 12) score++;
    59→  if (/[A-Z]/.test(password) && /[a-z]/.test(password)) score++;
    60→  if (/\d/.test(password)) score++;
    61→  if (/[^A-Za-z0-9]/.test(password)) score++;
    62→
    63→  // Normalize to 0–4
    64→  score = Math.min(4, score);
    65→
    66→  const labels = ['', 'Weak', 'Fair', 'Good', 'Strong'];
    67→  const colors = [
    68→    '',
    69→    'oklch(0.6 0.25 15)',   // red
    70→    'oklch(0.75 0.18 55)',  // amber
    71→    'oklch(0.85 0.18 90)',  // yellow-green
    72→    'oklch(0.72 0.19 155)', // green
    73→  ];
    74→
    75→  return { score, label: labels[score], color: colors[score] };
    76→}

    78→/** Email validation */
    79→function isValidEmail(email: string): boolean {
    80→  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
    81→}

   273→function PasswordStrengthIndicator({ password }: { password: string }) {
   274→  const strength = getPasswordStrength(password);
   275→  if (!password) return null;
   276→
   277→  const percent = (strength.score / 4) * 100;
   278→
   279→  return (
   280→    <div className="space-y-1.5">
   281→      <Progress
   282→        value={percent}
   283→        className="h-1.5"
   284→        style={
   285→          {
   286→            '--progress-color': strength.color,
   287→          } as React.CSSProperties
   288→        }
   289→      />
   290→      <p className="text-xs" style={{ color: strength.color }}>
   291→        {strength.label}
   292→      </p>
   293→    </div>
   294→  );
   295→}

   333→export function SignInPage({
   334→  onSignUpClick,
   335→  onForgotPasswordClick,
   336→  onSuccess,
   337→  onMagicLinkClick,
   338→  onOtpLoginClick,
   339→}: {
   340→  onSignUpClick?: () => void;
   341→  onForgotPasswordClick?: () => void;
   342→  onSuccess?: () => void;
   343→  onMagicLinkClick?: () => void;
   344→  onOtpLoginClick?: () => void;
   345→}) {
   346→  const { signIn, isLoading } = useAuth();
   347→  const [email, setEmail] = useState('');
   348→  const [password, setPassword] = useState('');
   349→  const [showPassword, setShowPassword] = useState(false);
   350→  const [errors, setErrors] = useState<Record<string, string>>({});
   351→  const [generalError, setGeneralError] = useState('');
   352→  const [authConfig, setAuthConfig] = useState<{ googleAvailable: boolean; loaded: boolean }>({ googleAvailable: false, loaded: false });
   353→
   354→  // Fetch auth config on mount to check Google OAuth availability
   355→  useEffect(() => {
   356→    fetch('/api/auth/config')
   357→      .then(res => res.ok ? res.json() : { googleAvailable: false })
   358→      .then(data => setAuthConfig({ googleAvailable: !!data.googleAvailable, loaded: true }))
   359→      .catch(() => setAuthConfig({ googleAvailable: false, loaded: true }));
   360→  }, []);
   361→
   362→  // Server-driven Google OAuth: client_id never touches frontend
   363→  const startGoogleOAuth = useCallback(async () => {
   364→    try {
   365→      const origin = typeof window !== 'undefined' ? window.location.origin : '';
   366→      const res = await fetch(`/api/auth/google/state?origin=${encodeURIComponent(origin)}`);
   367→      if (!res.ok) throw new Error('Failed to get OAuth state');
   368→      const data = await res.json();
   369→      if (data.authUrl) {
   370→        window.location.href = data.authUrl;
   371→      }
   372→    } catch (err) {
   373→      console.error('Google OAuth error:', err);
   374→      toast.error('Google sign-in is not available in this environment.');
   375→    }
   376→  }, []);
   377→
   378→  const validate = useCallback(() => {
   379→    const errs: Record<string, string> = {};
   380→    if (!email) errs.email = 'Email is required';
   381→    else if (!isValidEmail(email)) errs.email = 'Enter a valid email address';
   382→    if (!password) errs.password = 'Password is required';
   383→    else if (password.length < 6) errs.password = 'Password must be at least 6 characters';
   384→    return errs;
   385→  }, [email, password]);
   386→
   387→  const handleSubmit = useCallback(
   388→    async (e: React.FormEvent) => {
   389→      e.preventDefault();
   390→      setGeneralError('');
   391→      const validationErrors = validate();
   392→      setErrors(validationErrors);
   393→      if (Object.keys(validationErrors).length > 0) return;
   394→
   395→      const result = await signIn(email, password);
   396→      if (result.success) {
   397→        toast.success('Welcome back!');
   398→        onSuccess?.();
   399→      } else if (result.mfaRequired) {
   400→        toast.info('MFA verification required');
   401→        // Caller should handle MFA redirect via mfaRequired state
   402→        onSuccess?.();
   403→      } else {
   404→        setGeneralError(result.error || 'Sign in failed');
   405→        toast.error(result.error || 'Sign in failed');
   406→      }
   407→    },
   408→    [email, password, validate, signIn, onSuccess]
   409→  );
   410→
   411→  return (
   412→    <AuthLayout title="Welcome Back" description="Sign in to your AcquisitionOS account">
   413→      <Card className="glass-card">
   414→        <CardContent className="pt-6 space-y-4">
   415→          <AnimatePresence mode="wait">
   416→            {generalError && (
   417→              <motion.div
   418→                initial={{ opacity: 0, y: -6, height: 0 }}
   419→                animate={{ opacity: 1, y: 0, height: 'auto' }}
   420→                exit={{ opacity: 0, y: -6, height: 0 }}
   421→                className="rounded-lg border border-destructive/30 bg-destructive/5 p-3 text-sm text-destructive overflow-hidden"
   422→                role="alert"
   423→              >
   424→                <div className="flex items-center gap-2">
   425→                  <AlertTriangle className="h-4 w-4 shrink-0" />
   426→                  {generalError}
   427→                </div>
   428→              </motion.div>
   429→            )}
   430→          </AnimatePresence>
   431→
   432→          <form onSubmit={handleSubmit} className="space-y-4" noValidate>
   433→            {/* Email */}
   434→            <div className="space-y-2">
   435→              <Label htmlFor="signin-email" className="text-xs font-medium">
   436→                Email
   437→              </Label>
   438→              <div className={cn('relative transition-all duration-200 rounded-md', errors.email ? 'ring-1 ring-destructive/50' : email && isValidEmail(email) ? 'ring-1 ring-emerald-500/30' : '')}>
   439→                <Mail className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   440→                <Input
   441→                  id="signin-email"
   442→                  type="email"
   443→                  placeholder="you@example.com"
   444→                  className={cn('pl-9 h-10 border-0 bg-muted/50 focus-visible:bg-background transition-colors', errors.email && 'bg-destructive/5')}
   445→                  value={email}
   446→                  onChange={(e) => {
   447→                    setEmail(e.target.value);
   448→                    if (errors.email) setErrors((prev) => ({ ...prev, email: '' }));
   449→                  }}
   450→                  aria-invalid={!!errors.email}
   451→                  aria-describedby={errors.email ? 'signin-email-error' : undefined}
   452→                  autoComplete="email"
   453→                />
   454→                {email && isValidEmail(email) && !errors.email && (
   455→                  <Check className="absolute right-3 top-1/2 -translate-y-1/2 h-4 w-4 text-emerald-500" />
