   214→/** Validate password strength */
   215→export function validatePasswordStrength(password: string): {
   216→  valid: boolean;
   217→  errors: string[];
   218→} {
   219→  const errors: string[] = [];
   220→  if (password.length < 8) errors.push('Password must be at least 8 characters');
   221→  if (password.length > 128) errors.push('Password must be less than 128 characters');
   222→  if (!/[A-Z]/.test(password)) errors.push('Password must contain at least one uppercase letter');
   223→  if (!/[a-z]/.test(password)) errors.push('Password must contain at least one lowercase letter');
   224→  if (!/[0-9]/.test(password)) errors.push('Password must contain at least one number');
   225→  if (!/[^A-Za-z0-9]/.test(password))
   226→    errors.push('Password must contain at least one special character');
   227→  return { valid: errors.length === 0, errors };
   228→}

   242→/** Constant-time string comparison to prevent timing attacks */
   243→export function secureCompare(a: string, b: string): boolean {
   244→  if (typeof a !== 'string' || typeof b !== 'string') return false;
   245→  const bufA = Buffer.from(a, 'utf-8');
   246→  const bufB = Buffer.from(b, 'utf-8');
   247→  // If lengths differ, still compare bufB with itself to consume the same time,
   248→  // then return false — this prevents leaking length information via timing
   249→  if (bufA.length !== bufB.length) {
   250→    crypto.timingSafeEqual(bufB, bufB);
   251→    return false;
   252→  }
   253→  try {
   254→    return crypto.timingSafeEqual(bufA, bufB);
   255→  } catch {
   256→    return false;
   257→  }
   258→}

   276→    data: {
   277→      otpAttemptCount: newCount,
   278→      otpLockedUntil: lockUntil,
   279→    },
   280→  });
   281→
   282→  return { locked, attempts: newCount };

   300→/** Create a new user session in the database */
   301→export async function createSession(params: {
   302→  userId: string;
   303→  refreshToken: string;
   304→  deviceInfo?: string;
   305→  ipAddress?: string;
   306→  userAgent?: string;
   307→}): Promise<void> {
   308→  const expiresAt = new Date(Date.now() + 30 * 24 * 60 * 60 * 1000); // 30 days
   309→
   310→  // Pre-emptively delete any existing session with the same refreshToken
   311→  // to avoid P2002 unique constraint errors on race conditions or re-logins.
   312→  try {
   313→    await db.userSession.deleteMany({
   314→      where: { refreshToken: params.refreshToken },
   315→    });
   316→  } catch {
   317→    // Ignore errors — the session may not exist
   318→  }
   319→
   320→  try {
   321→    await db.userSession.create({
   322→      data: {
   323→        userId: params.userId,
   324→        refreshToken: params.refreshToken,
   325→        deviceInfo: params.deviceInfo || null,
   326→        ipAddress: params.ipAddress || null,
   327→        userAgent: params.userAgent || null,
   328→        expiresAt,
   329→        isRevoked: false,
   330→      },
   331→    });
   332→  } catch (error: unknown) {
   333→    // Handle P2002 (unique constraint violation) gracefully.
   334→    // This can still happen in a race condition where another request
   335→    // inserted the same refreshToken between our deleteMany and create.
   336→    if (
   337→      error &&
   338→      typeof error === 'object' &&
   339→      'code' in error &&
   340→      (error as { code: string }).code === 'P2002'
   341→    ) {
   342→      // Delete the conflicting session and retry once
   343→      await db.userSession.deleteMany({
   344→        where: { refreshToken: params.refreshToken },
   345→      });
   346→
   347→      await db.userSession.create({
   348→        data: {
   349→          userId: params.userId,
   350→          refreshToken: params.refreshToken,
   351→          deviceInfo: params.deviceInfo || null,
   352→          ipAddress: params.ipAddress || null,
   353→          userAgent: params.userAgent || null,
   354→          expiresAt,
   355→          isRevoked: false,
   356→        },
   357→      });
   358→    } else {
   359→      throw error;
   360→    }
   361→  }
   362→}

   469→// ===== MFA / TOTP FUNCTIONS =====
   470→
   471→const TOTP_PERIOD = 30; // seconds per time step
   472→const TOTP_DIGITS = 6;
   473→const TOTP_WINDOW = 1; // allow ±1 time step for clock drift
   474→
   475→// Base32 alphabet (RFC 4648)
   476→const BASE32_CHARS = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ234567';
   477→
   478→/** Encode a buffer as Base32 string */
   479→function base32Encode(buffer: Buffer): string {
   480→  let bits = '';
   481→  for (const byte of buffer) {
   482→    bits += byte.toString(2).padStart(8, '0');
   483→  }
   484→  let result = '';
   485→  for (let i = 0; i + 5 <= bits.length; i += 5) {
   486→    result += BASE32_CHARS[parseInt(bits.substring(i, i + 5), 2)];
   487→  }
   488→  return result;
   489→}
   490→
   491→/** Decode a Base32 string to a buffer */
   492→function base32Decode(str: string): Buffer {
   493→  const cleaned = str.toUpperCase().replace(/[^A-Z2-7]/g, '');
   494→  let bits = '';
   495→  for (const char of cleaned) {
   496→    const val = BASE32_CHARS.indexOf(char);
   497→    if (val === -1) continue;
   498→    bits += val.toString(2).padStart(5, '0');
   499→  }
   500→  const bytes: number[] = [];
   501→  for (let i = 0; i + 8 <= bits.length; i += 8) {
   502→    bytes.push(parseInt(bits.substring(i, i + 8), 2));
   503→  }
   504→  return Buffer.from(bytes);
   505→}

   508→export function generateTotpSecret(): string {
   509→  return base32Encode(crypto.randomBytes(20));
   510→}

   512→/** Generate an otpauth:// URI for QR code generation */
   513→export function generateTotpUri(params: {
   514→  secret: string;
   515→  label: string;
   516→  issuer: string;
   517→}): string {
   518→  const encodedLabel = encodeURIComponent(params.label);
   519→  const encodedIssuer = encodeURIComponent(params.issuer);
   520→  return `otpauth://totp/${encodedIssuer}:${encodedLabel}?secret=${params.secret}&issuer=${encodedIssuer}&algorithm=SHA1&digits=${TOTP_DIGITS}&period=${TOTP_PERIOD}`;
   521→}
   522→
   523→/** Calculate TOTP code for a given secret and time */
   524→function calculateTotp(secret: string, time: number): string {
   525→  const timeStep = Math.floor(time / TOTP_PERIOD);
   526→  const timeBuffer = Buffer.alloc(8);
   527→  timeBuffer.writeUInt32BE(0, 0);
   528→  timeBuffer.writeUInt32BE(timeStep, 4);
   529→
   530→  const key = base32Decode(secret);
   531→  const hmac = crypto.createHmac('sha1', key);
   532→  hmac.update(timeBuffer);
   533→  const hmacResult = hmac.digest();
   534→
   535→  // Dynamic truncation
   536→  const offset = hmacResult[hmacResult.length - 1] & 0x0f;
   537→  const code =
   538→    ((hmacResult[offset] & 0x7f) << 24) |
   539→    ((hmacResult[offset + 1] & 0xff) << 16) |
   540→    ((hmacResult[offset + 2] & 0xff) << 8) |
   541→    (hmacResult[offset + 3] & 0xff);
   542→
   543→  return (code % Math.pow(10, TOTP_DIGITS)).toString().padStart(TOTP_DIGITS, '0');
   544→}
   545→
   546→/** Verify a TOTP code against a secret with clock drift tolerance */
   547→export function verifyTotpCode(secret: string, token: string): boolean {
   548→  if (!/^\d{6}$/.test(token)) return false;
   549→
   550→  const now = Math.floor(Date.now() / 1000);
   551→  // Check current time step and ±window steps
   552→  for (let i = -TOTP_WINDOW; i <= TOTP_WINDOW; i++) {
   553→    const checkTime = now + i * TOTP_PERIOD;
   554→    const expected = calculateTotp(secret, checkTime);
   555→    if (crypto.timingSafeEqual(Buffer.from(token), Buffer.from(expected))) {
   556→      return true;
   557→    }
   558→  }
   559→  return false;
   560→}

   611→/**
   612→ * Map a Prisma User record to an AuthUser.
   613→ * Handles field name differences (e.g., `avatar` → `avatarUrl`)
   614→ * and resolves MFA status from the MfaConfig relation.
   615→ * Resolves plan from the active subscription if available, falling back
   616→ * to the User.plan field, and ultimately to 'free'.
   617→ */
   618→async function mapUserToAuthUser(
   619→  user: {
   620→    id: string;
   621→    email: string;
   622→    name: string | null;
   623→    role: string;
   624→    plan: string;
   625→    orgId: string | null;
   626→    emailVerified: boolean;
   627→    avatar: string | null;
   628→    mfaConfig: { isEnabled: boolean } | null;
   629→    subscriptions: { plan: string; status: string }[];
   630→  }
   631→): Promise<AuthUser> {
   632→  // Resolve the plan from the active subscription (trialing or active),
   633→  // falling back to the User.plan field, then to 'free'.
   634→  const activeSubscription = user.subscriptions.find(
   635→    (s) => s.status === 'active' || s.status === 'trialing'
   636→  );
   637→  const resolvedPlan = activeSubscription?.plan || user.plan || 'free';
   638→
   639→  return {
   640→    id: user.id,
   641→    email: user.email,
   642→    name: user.name || '',
   643→    role: user.role,
   644→    plan: resolvedPlan,
   645→    orgId: user.orgId,
   646→    emailVerified: user.emailVerified,
   647→    mfaEnabled: user.mfaConfig?.isEnabled ?? false,
   648→    avatarUrl: user.avatar,
   649→  };
   650→}
   651→
   652→/** Get the authenticated user from a request, or null */
   653→export async function getAuthUser(request: NextRequest): Promise<AuthUser | null> {
   654→  try {
   655→    // Try Bearer token first
   656→    const bearerToken = extractBearerToken(request);
   657→    if (bearerToken) {
   658→      const payload = verifyToken(bearerToken);
   659→      if (payload && payload.type === 'access') {
   660→        const user = await db.user.findUnique({
   661→          where: { id: payload.sub },
   662→          include: {
   663→            mfaConfig: { select: { isEnabled: true } },
   664→            subscriptions: {
   665→              where: { status: { in: ['active', 'trialing'] } },
   666→              select: { plan: true, status: true },
   667→              orderBy: { createdAt: 'desc' },
   668→              take: 1,
   669→            },
   670→          },
   671→        });
   672→        if (user) {
   673→          return await mapUserToAuthUser(user);
   674→        }
   675→      }
   676→    }
   677→
   678→    // Try access_token cookie
   679→    const accessTokenCookie = request.cookies.get('access_token')?.value;
   680→    if (accessTokenCookie) {
   681→      const payload = verifyToken(accessTokenCookie);
   682→      if (payload && payload.type === 'access') {
   683→        const user = await db.user.findUnique({
   684→          where: { id: payload.sub },
   685→          include: {
   686→            mfaConfig: { select: { isEnabled: true } },
   687→            subscriptions: {
   688→              where: { status: { in: ['active', 'trialing'] } },
   689→              select: { plan: true, status: true },
   690→              orderBy: { createdAt: 'desc' },
   691→              take: 1,
   692→            },
   693→          },
   694→        });
   695→        if (user) {
   696→          return await mapUserToAuthUser(user);
   697→        }
   698→      }
   699→    }
   700→
   701→    return null;
   702→  } catch {
   703→    return null;
   704→  }
   705→}

   707→/** Require authentication — returns user or throws a 401 response */
   708→export async function requireAuth(request: NextRequest): Promise<AuthUser> {
   709→  const user = await getAuthUser(request);
   710→  if (!user) {
   711→    throw new AuthError('Authentication required', 401);
   712→  }
   713→  return user;
   714→}

   740→/** Extract IP address from request */
   741→export function getClientIp(request: NextRequest): string {
   742→  const forwarded = request.headers.get('x-forwarded-for');
   743→  if (forwarded) return forwarded.split(',')[0].trim();
   744→  const realIp = request.headers.get('x-real-ip');
   745→  if (realIp) return realIp;
   746→  return 'unknown';
   747→}

   750→export function getUserAgent(request: NextRequest): string {
   751→  return request.headers.get('user-agent') || 'unknown';

   754→// ===== SUSPICIOUS LOGIN DETECTION =====
   755→
   756→/**
   757→ * Detect if a login is from a new IP or device compared to the user's
   758→ * recent login history. Returns true if the login looks suspicious.
   759→ */
   760→export async function detectSuspiciousLogin(params: {
   761→  userId: string;
   762→  ip: string;
   763→  userAgent: string;
   764→}): Promise<boolean> {
   765→  // Get recent login history for this user (last 30 days)
   766→  const recentLogins = await db.loginHistory.findMany({
   767→    where: {
   768→      userId: params.userId,
   769→      success: true,
   770→      createdAt: { gt: new Date(Date.now() - 30 * 24 * 60 * 60 * 1000) },
   771→    },
   772→    orderBy: { createdAt: 'desc' },
   773→    take: 10,
   774→  });
   775→
   776→  // Check for new IP
   777→  const knownIps = new Set(recentLogins.map(l => l.ip).filter(Boolean));
   778→  const isNewIp = !knownIps.has(params.ip) && knownIps.size > 0;
   779→
   780→  // Check for new device (user agent)
   781→  const knownUserAgents = new Set(recentLogins.map(l => l.userAgent).filter(Boolean));
   782→  const isNewDevice = !knownUserAgents.has(params.userAgent) && knownUserAgents.size > 0;
   783→
   784→  return isNewIp || isNewDevice;
   785→}

   787→/**
   788→ * Send a security alert to the user about a suspicious login.
   789→ * In production, this would send an email. For now, it logs and
   790→ * records the event in the audit log.
   791→ */
   792→export async function sendSecurityAlert(params: {
   793→  userId: string;
   794→  email: string;
   795→  name: string;
   796→  event: string;
   797→  ip: string;
   798→  userAgent: string;
   799→}): Promise<void> {
   800→  // In production, send an email to the user
   801→  console.log(`[SECURITY ALERT] ${params.event} for ${params.email} from IP ${params.ip}`);
   802→
   803→  // Send security alert email (non-blocking)
   804→  try {
   805→    const { sendSecurityAlertEmail, isEmailServiceConfigured } = await import('./email');
   806→    const { shouldBypassEmail } = await import('./feature-flags');
   807→    if (isEmailServiceConfigured() && !shouldBypassEmail()) {
   808→      await sendSecurityAlertEmail(params.email, params.name, params.event, params.ip, params.userAgent);
   809→    }
   810→  } catch {
   811→    // Never block the main flow
   812→  }
   813→
   814→  await logAuthEvent({
   815→    userId: params.userId,
   816→    action: 'suspicious_login',
   817→    details: `${params.event} from IP ${params.ip}`,
   818→    ipAddress: params.ip,
   819→    userAgent: params.userAgent,
   820→  });
   821→}
