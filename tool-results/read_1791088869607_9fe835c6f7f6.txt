    90→function getRelativeTime(dateStr: string): string {
    91→  const date = new Date(dateStr);
    92→  const now = new Date();
    93→  const diffMs = now.getTime() - date.getTime();
    94→  const diffSecs = Math.floor(diffMs / 1000);
    95→  const diffMins = Math.floor(diffSecs / 60);
    96→  const diffHours = Math.floor(diffMins / 60);
    97→  const diffDays = Math.floor(diffHours / 24);
    98→  if (diffSecs < 60) return 'Just now';
    99→  if (diffMins < 60) return `${diffMins}m ago`;
   100→  if (diffHours < 24) return `${diffHours}h ago`;
   101→  if (diffDays === 1) return 'Yesterday';
   102→  if (diffDays < 7) return `${diffDays}d ago`;
   103→  return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
   104→}

   121→function getWebsiteQualityBadge(lead: Lead) {
   122→  if (!lead.hasWebsite) return <Badge className="bg-red-500/15 text-red-400 border-red-500/20">No Website</Badge>;
   123→  const q = lead.websiteQuality?.toLowerCase();
   124→  if (q === 'poor') return <Badge className="bg-orange-500/15 text-orange-400 border-orange-500/20">Poor</Badge>;
   125→  if (q === 'average') return <Badge className="bg-yellow-500/15 text-yellow-400 border-yellow-500/20">Average</Badge>;
   126→  if (q === 'good' || q === 'excellent') return <Badge className="bg-emerald-500/15 text-emerald-400 border-emerald-500/20">Good</Badge>;
   127→  return <Badge className="bg-slate-500/15 text-slate-400 border-slate-500/20">Unknown</Badge>;
   128→}

   130→// ─── Notes Tab Content ───────────────────────────────────
   131→function NotesTab({ leadId }: { leadId: string }) {
   132→  const queryClient = useQueryClient();
   133→  const [newNote, setNewNote] = useState('');
   134→  const [adding, setAdding] = useState(false);
   135→
   136→  const { data: notesResult, isLoading } = useQuery({
   137→    queryKey: ['notes', leadId],
   138→    queryFn: async () => {
   139→      const res = await fetch(`/api/leads/${leadId}/notes`);
   140→      if (!res.ok) throw new Error('Failed to fetch notes');
   141→      return res.json();
   142→    },
   143→    enabled: !!leadId,
   144→  });
   145→
   146→  const notes = notesResult?.notes ?? [];
   147→
   148→  const handleAddNote = useCallback(async () => {
   149→    if (!newNote.trim()) return;
   150→    setAdding(true);
   151→    try {
   152→      const res = await fetch(`/api/leads/${leadId}/notes`, {
   153→        method: 'POST',
   154→        headers: { 'Content-Type': 'application/json' },
   155→        body: JSON.stringify({ content: newNote.trim() }),
   156→      });
   157→      if (!res.ok) throw new Error('Failed to add note');
   158→      setNewNote('');
   159→      queryClient.invalidateQueries({ queryKey: ['notes', leadId] });
   160→      queryClient.invalidateQueries({ queryKey: ['leads'] });
   161→      toast.success('Note added');
   162→    } catch {
   163→      toast.error('Failed to add note');
   164→    } finally {
   165→      setAdding(false);
   166→    }
   167→  }, [leadId, newNote, queryClient]);
   168→
   169→  if (isLoading) {
   170→    return (
   171→      <div className="space-y-3">
   172→        {Array.from({ length: 3 }).map((_, i) => (
   173→          <Skeleton key={i} className="h-16 w-full rounded-lg" />
   174→        ))}
   175→      </div>
   176→    );
   177→  }
   178→
   179→  return (
   180→    <div className="space-y-4">
   181→      {/* Add Note Form */}
   182→      <div className="space-y-2">
   183→        <Textarea
   184→          placeholder="Add a note..."
   185→          value={newNote}
   186→          onChange={(e) => setNewNote(e.target.value)}
   187→          className="min-h-[80px] border-primary/20 focus-visible:ring-primary/30"
   188→        />
   189→        <div className="flex justify-end">
   190→          <Button
   191→            size="sm"
   192→            onClick={handleAddNote}
   193→            disabled={!newNote.trim() || adding}
   194→            className="gap-1.5"
   195→          >
   196→            {adding ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Plus className="h-3.5 w-3.5" />}
   197→            Add Note
   198→          </Button>
   199→        </div>
   200→      </div>
   201→
   202→      <Separator />
   203→
   204→      {/* Notes List */}
   205→      {notes.length > 0 ? (
   206→        <div className="space-y-3">
   207→          {notes.map((note: { id: string; content: string; createdAt: string; pinned?: boolean; user?: { name?: string } }) => (
   208→            <Card key={note.id} className={cn('border', note.pinned && 'border-primary/20')}>
   209→              <CardContent className="p-3 space-y-2">
   210→                <div className="flex items-start justify-between gap-2">
   211→                  <p className="text-sm leading-relaxed flex-1">{note.content}</p>
   212→                  {note.pinned && <Badge variant="outline" className="text-[9px] shrink-0">Pinned</Badge>}
   213→                </div>
   214→                <div className="flex items-center gap-2 text-xs text-muted-foreground">
   215→                  <Clock className="h-3 w-3" />
   216→                  {getRelativeTime(note.createdAt)}
   217→                  {note.user?.name && (
   218→                    <>
   219→                      <span>·</span>
   220→                      <span>{note.user.name}</span>
   221→                    </>
   222→                  )}
   223→                </div>
   224→              </CardContent>
   225→            </Card>
   226→          ))}
   227→        </div>
   228→      ) : (
   229→        <div className="text-center py-6 text-muted-foreground">
   230→          <StickyNote className="h-8 w-8 mx-auto mb-2 opacity-30" />
   231→          <p className="text-sm">No notes yet</p>
   232→        </div>
   233→      )}
   234→    </div>
   235→  );
   236→}

   239→function ActivityTab({ leadId }: { leadId: string }) {
   240→  const { data: activities, isLoading } = useQuery({
   241→    queryKey: ['activities', leadId],
   242→    queryFn: () => fetchLeadActivities(leadId),
   243→    enabled: !!leadId,
   244→  });
   245→
   246→  if (isLoading) {
   247→    return (
   248→      <div className="space-y-3">
   249→        {Array.from({ length: 4 }).map((_, i) => (
   250→          <Skeleton key={i} className="h-12 w-full rounded-lg" />
   251→        ))}
   252→      </div>
   253→    );
   254→  }
   255→
   256→  if (!activities || activities.length === 0) {
   257→    return (
   258→      <div className="text-center py-6 text-muted-foreground">
   259→        <Activity className="h-8 w-8 mx-auto mb-2 opacity-30" />
   260→        <p className="text-sm">No activity recorded</p>
   261→      </div>
   262→    );
   263→  }
   264→
   265→  return (
   266→    <div className="space-y-0">
   267→      {activities.map((act, idx) => {
   268→        const { icon: Icon, bg, text } = getActivityIcon(act.type);
   269→        const isLast = idx === activities.length - 1;
   270→        return (
   271→          <div key={act.id} className="flex items-start gap-3 relative">
   272→            {/* Timeline line */}
   273→            {!isLast && (
   274→              <div className="absolute left-[11px] top-7 bottom-0 w-px bg-border/50" />
   275→            )}
   276→            <div className={cn('rounded-lg p-1.5 shrink-0 z-10', bg)}>
   277→              <Icon className={cn('h-3.5 w-3.5', text)} />
   278→            </div>
   279→            <div className="flex-1 min-w-0 pb-3">
   280→              <p className="text-sm">{act.description}</p>
   281→              <p className="text-[10px] text-muted-foreground mt-0.5">{getRelativeTime(act.createdAt)}</p>
   282→            </div>
   283→          </div>
   284→        );
   285→      })}
   286→    </div>
   287→  );
   288→}
