    90→    90→function getRelativeTime(dateStr: string): string {
    91→    91→  const date = new Date(dateStr);
    92→    92→  const now = new Date();
    93→    93→  const diffMs = now.getTime() - date.getTime();
    94→    94→  const diffSecs = Math.floor(diffMs / 1000);
    95→    95→  const diffMins = Math.floor(diffSecs / 60);
    96→    96→  const diffHours = Math.floor(diffMins / 60);
    97→    97→  const diffDays = Math.floor(diffHours / 24);
    98→    98→  if (diffSecs < 60) return 'Just now';
    99→    99→  if (diffMins < 60) return `${diffMins}m ago`;
   100→   100→  if (diffHours < 24) return `${diffHours}h ago`;
   101→   101→  if (diffDays === 1) return 'Yesterday';
   102→   102→  if (diffDays < 7) return `${diffDays}d ago`;
   103→   103→  return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
   104→   104→}
