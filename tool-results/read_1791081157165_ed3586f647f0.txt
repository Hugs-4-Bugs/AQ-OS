   217→// ── OTP Login page ────────────────────────────────────────────
   218→function OtpLoginPage({ onBackToSignIn }: { onBackToSignIn: () => void }) {
   219→  const [email, setEmail] = useState('');
   220→  const [otp, setOtp] = useState('');
   221→  const [step, setStep] = useState<'request' | 'verify'>('request');
   222→  const [loading, setLoading] = useState(false);
   223→  const [error, setError] = useState('');
   224→  const [resendCooldown, setResendCooldown] = useState(0);
   225→  const [emailPreviewUrl, setEmailPreviewUrl] = useState<string | undefined>();
   226→  const [emailProvider, setEmailProvider] = useState<string | undefined>();
   227→  const [deliveryIssue, setDeliveryIssue] = useState(false);
   228→  const [deliveryMessage, setDeliveryMessage] = useState('');
   229→  const [fallbackReason, setFallbackReason] = useState<string | undefined>();
   230→  // DEV-ONLY (sandbox): the OTP surfaced in-place when the server cannot
   231→  // send emails. Never set in production.
   232→  const [devDelivery, setDevDelivery] = useState<DevDeliveryPayload | undefined>();
   233→  // P5: "Remember me for 30 days" — wired to the OTP login session
   234→  const [rememberMe, setRememberMe] = useState(false);
   235→  const cooldownRef = useRef<ReturnType<typeof setInterval> | undefined>(undefined);
   236→
   237→  // Cooldown timer for resend
   238→  useEffect(() => {
   239→    if (resendCooldown <= 0) return;
   240→    cooldownRef.current = setInterval(() => {
   241→      setResendCooldown((prev) => {
   242→        if (prev <= 1) {
   243→          clearInterval(cooldownRef.current);
   244→          return 0;
   245→        }
   246→        return prev - 1;
   247→      });
   248→    }, 1000);
   249→    return () => clearInterval(cooldownRef.current);
   250→  }, [resendCooldown]);
   251→
   252→  const handleRequestOtp = useCallback(
   253→    async (e: React.FormEvent) => {
   254→      e.preventDefault();
   255→      if (!email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   256→        setError('Enter a valid email address');
   257→        return;
   258→      }
   259→      setError('');
   260→      setLoading(true);
   261→      try {
   262→        const res = await fetch('/api/auth/otp/request', {
   263→          method: 'POST',
   264→          headers: { 'Content-Type': 'application/json' },
   265→          body: JSON.stringify({ email }),
   266→        });
   267→        const data = await res.json();
   268→        if (res.ok) {
   269→          setStep('verify');
   270→          setResendCooldown(60);
   271→          // Surface the Ethereal preview URL so the user can open the email
   272→          // and read the OTP. If no preview URL is returned (real SMTP /
   273→          // Resend), the user must check their actual inbox.
   274→          setEmailPreviewUrl(data.emailPreviewUrl);
   275→          setEmailProvider(data.emailProvider);
   276→          setFallbackReason(data.fallbackReason);
   277→          // DEV-ONLY: capture the in-place delivered code (sandbox mode).
   278→          setDevDelivery(parseDevDelivery(data));
   279→          // Surface delivery failures (e.g. Gmail daily quota exceeded) so the
   280→          // user knows the OTP will NOT arrive and doesn't wait forever.
   281→          setDeliveryIssue(!!data.deliveryIssue);
   282→          setDeliveryMessage(data.deliveryMessage || '');
   283→          if (data.deliveryIssue && !data.devDelivery) {
   284→            toast.error('Email delivery failed — see details below');
   285→          } else if (data.devDelivery) {
   286→            toast.info('Dev mode: email delivery not configured — your login code is shown below.');
   287→          } else if (data.fallbackReason) {
   288→            toast.warning('Gmail limit reached — OTP delivered to preview inbox. Click the preview link below.');
   289→          } else {
   290→            toast.success('If an account exists for this email, you will receive a sign-in code.');
   291→          }
   292→        } else {
   293→          setError(data.error || 'Failed to send OTP');
   294→          toast.error(data.error || 'Failed to send OTP');
   295→        }
   296→      } catch {
   297→        setError('Network error. Please try again.');
   298→      } finally {
   299→        setLoading(false);
   300→      }
   301→    },
   302→    [email]
   303→  );
   304→
   305→  const handleVerifyOtp = useCallback(async () => {
   306→    if (otp.length !== 6) {
   307→      setError('Enter the 6-digit code');
   308→      return;
   309→    }
   310→    setError('');
   311→    setLoading(true);
   312→    try {
   313→      const res = await fetch('/api/auth/otp/verify', {
   314→        method: 'POST',
   315→        headers: { 'Content-Type': 'application/json' },
   316→        credentials: 'include',
   317→        body: JSON.stringify({ email, otp, rememberMe }),
   318→      });
   319→      const data = await res.json();
   320→      if (res.ok) {
   321→        toast.success('Welcome back!');
   322→        // Auth cookies are set by the API, refresh auth state
   323→        window.location.reload();
   324→      } else {
   325→        setError(data.error || 'Invalid OTP');
   326→        toast.error(data.error || 'Verification failed');
   327→      }
   328→    } catch {
   329→      setError('Network error. Please try again.');
   330→    } finally {
   331→      setLoading(false);
   332→    }
   333→  }, [email, otp, rememberMe]);
   334→
   335→  return (
   336→    <AuthLayout
   337→      title="Sign in with OTP"
   338→      description={step === 'request' ? 'Enter your email to receive a login code' : `Enter the 6-digit code sent to ${email}`}
   339→      showBack
   340→      onBack={onBackToSignIn}
   341→    >
   342→      <Card className="glass-card">
   343→        <CardContent className="pt-6 space-y-4">
   344→          <AnimatePresence mode="wait">
   345→            {error && (
   346→              <motion.div
   347→                initial={{ opacity: 0, y: -6, height: 0 }}
   348→                animate={{ opacity: 1, y: 0, height: 'auto' }}
   349→                exit={{ opacity: 0, y: -6, height: 0 }}
   350→                className="rounded-lg border border-destructive/30 bg-destructive/5 p-3 text-sm text-destructive overflow-hidden"
   351→                role="alert"
   352→              >
   353→                <div className="flex items-center gap-2">
   354→                  <AlertTriangle className="h-4 w-4 shrink-0" />
   355→                  {error}
   356→                </div>
   357→              </motion.div>
   358→            )}
   359→          </AnimatePresence>
   360→
   361→          <AnimatePresence mode="wait">
   362→            {step === 'request' ? (
   363→              <motion.form
   364→                key="request"
   365→                onSubmit={handleRequestOtp}
   366→                className="space-y-4"
   367→                noValidate
   368→                initial={{ opacity: 0, x: -20 }}
   369→                animate={{ opacity: 1, x: 0 }}
   370→                exit={{ opacity: 0, x: 20 }}
   371→                transition={{ duration: 0.2 }}
   372→              >
   373→                <div className="space-y-2">
   374→                  <Label htmlFor="otp-email" className="text-xs font-medium">
   375→                    Email
   376→                  </Label>
   377→                  <div className="relative">
   378→                    <Mail className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   379→                    <Input
   380→                      id="otp-email"
   381→                      type="email"
   382→                      placeholder="you@example.com"
   383→                      className="pl-9 h-10 border-0 bg-muted/50 focus-visible:bg-background transition-colors"
   384→                      value={email}
   385→                      onChange={(e) => { setEmail(e.target.value); if (error) setError(''); }}
   386→                      aria-invalid={!!error}
   387→                      autoComplete="email"
   388→                    />
   389→                  </div>
   390→                </div>
   391→                {/* P5: Remember me — controls the session created on verify */}
   392→                <div className="flex items-center gap-2">
   393→                  <Checkbox
   394→                    id="otp-remember"
   395→                    checked={rememberMe}
   396→                    onCheckedChange={(checked) => setRememberMe(checked === true)}
   397→                  />
   398→                  <Label htmlFor="otp-remember" className="text-xs text-muted-foreground cursor-pointer">
   399→                    Remember me for 30 days
   400→                  </Label>
   401→                </div>
   402→                <Button
   403→                  type="submit"
   404→                  disabled={loading}
   405→                  className="w-full h-10 btn-ripple"
   406→                >
   407→                  {loading ? (
   408→                    <>
   409→                      <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   410→                      Sending...
   411→                    </>
   412→                  ) : (
   413→                    <>
   414→                      <Key className="mr-2 h-4 w-4" />
   415→                      Send Login Code
   416→                    </>
   417→                  )}
   418→                </Button>
   419→              </motion.form>
   420→            ) : (
   421→              <motion.div
   422→                key="verify"
   423→                className="space-y-4"
   424→                initial={{ opacity: 0, x: 20 }}
   425→                animate={{ opacity: 1, x: 0 }}
   426→                exit={{ opacity: 0, x: -20 }}
   427→                transition={{ duration: 0.2 }}
   428→              >
   429→                {deliveryIssue && (
   430→                  <motion.div
   431→                    initial={{ opacity: 0, y: -6, height: 0 }}
   432→                    animate={{ opacity: 1, y: 0, height: 'auto' }}
   433→                    className="rounded-lg border border-amber-500/40 bg-amber-500/10 p-3 text-sm text-amber-700 dark:text-amber-400 overflow-hidden"
   434→                    role="alert"
   435→                  >
   436→                    <div className="flex items-start gap-2">
   437→                      <AlertTriangle className="h-4 w-4 shrink-0 mt-0.5" />
   438→                      <div>

   440→                        <p className="text-xs leading-relaxed">{deliveryMessage}</p>
   441→                      </div>
   442→                    </div>
   443→                  </motion.div>
   444→                )}
   445→                <DevDeliveryNotice
   446→                  delivery={devDelivery}

   449→                <EmailPreviewNotice

   460→                      <InputOTPSlot index={1} className="h-12 w-12 text-lg" />
