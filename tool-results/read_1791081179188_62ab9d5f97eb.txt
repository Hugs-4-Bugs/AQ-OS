   127→function SidebarNav({
   128→  activeTab,
   129→  onTabChange,
   130→  className,
   131→  orientation = 'vertical',
   132→  collapsed = false,
   133→}: {
   134→  activeTab: TabId;
   135→  onTabChange: (tab: TabId) => void;
   136→  className?: string;
   137→  orientation?: 'vertical' | 'horizontal';
   138→  collapsed?: boolean;
   139→}) {
   140→  return (
   141→    <nav role="navigation" aria-label="Main navigation" className={cn('flex gap-1', orientation === 'vertical' ? 'flex-col' : 'flex-row', className)}>
   142→      {NAV_ITEMS.map((item) => {
   143→        const Icon = item.icon;
   144→        const isActive = activeTab === item.id;
   145→        const button = (
   146→          <button
   147→            key={item.id}
   148→            onClick={() => onTabChange(item.id)}
   149→            aria-label={collapsed && orientation === 'vertical' ? item.label : undefined}
   150→            className={cn(
   151→              'relative flex items-center rounded-lg text-sm font-medium transition-all duration-200',
   152→              'hover:bg-accent hover:text-accent-foreground',
   153→              'focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring',
   154→              collapsed && orientation === 'vertical' ? 'justify-center px-2 py-2.5' : 'gap-3 px-3 py-2.5',
   155→              isActive
   156→                ? 'bg-primary/10 text-primary font-semibold'
   157→                : 'text-muted-foreground',
   158→              orientation === 'horizontal' && 'flex-col gap-1 px-2 py-1.5 text-xs'
   159→            )}
   160→          >
   161→            <Icon className="shrink-0 h-4 w-4" />
   162→            {(!collapsed || orientation === 'horizontal') && (
   163→              <span className={cn(orientation === 'horizontal' && 'hidden sm:inline')}>{item.label}</span>
   164→            )}
   165→            {isActive && orientation === 'vertical' && !collapsed && (
   166→              <motion.div
   167→                layoutId="sidebar-indicator"
   168→                className="absolute left-0 h-6 w-1 rounded-r-full bg-primary"
   169→                transition={{ type: 'spring', stiffness: 350, damping: 30 }}
   170→              />
   171→            )}
   172→            {isActive && orientation === 'vertical' && collapsed && (
   173→              <motion.div
   174→                layoutId="sidebar-indicator-collapsed"
   175→                className="absolute left-0 h-6 w-1 rounded-r-full bg-primary"
   176→                transition={{ type: 'spring', stiffness: 350, damping: 30 }}
   177→              />
   178→            )}
   179→            {isActive && orientation === 'horizontal' && (
   180→              <motion.div
   181→                layoutId="mobile-indicator"
   182→                className="absolute -bottom-1 h-0.5 w-6 rounded-full bg-primary"
   183→                transition={{ type: 'spring', stiffness: 350, damping: 30 }}
   184→              />
   185→            )}
   186→          </button>
   187→        );
   188→
   189→        // Collapsed desktop rail: labels are hidden, so expose the destination
   190→        // via a right-side tooltip. Radix tooltips also open on keyboard focus,
   191→        // which keeps the rail fully keyboard accessible.
   192→        if (collapsed && orientation === 'vertical') {
   193→          return (
   194→            <Tooltip key={item.id}>
   195→              <TooltipTrigger asChild>
   196→                {button}
   197→              </TooltipTrigger>
   198→              <TooltipContent side="right" sideOffset={8}>
   199→                {item.label}
   200→              </TooltipContent>
   201→            </Tooltip>
   202→          );
   203→        }
   204→
   205→        return button;
   206→      })}
   207→    </nav>
   208→  );
   209→}

   211→/* ===== Live Clock Component ===== */
   212→function LiveClock() {
   213→  const [time, setTime] = useState('');
   214→
   215→  useEffect(() => {
   216→    const update = () => {
   217→      const now = new Date();
   218→      setTime(
   219→        now.toLocaleTimeString('en-US', {
   220→          hour12: false,
   221→          hour: '2-digit',
   222→          minute: '2-digit',
   223→          second: '2-digit',
   224→        })
   225→      );
   226→    };
   227→    update();
   228→    const interval = setInterval(update, 1000);
   229→    return () => clearInterval(interval);
   230→  }, []);
   231→
   232→  return (
   233→    <div className="flex items-center gap-1.5 text-xs text-muted-foreground">
   234→      <Clock className="h-3 w-3" />
   235→      <span className="font-mono tabular-nums">{time || '--:--:--'}</span>
   236→    </div>
   237→  );
   238→}

   319→  // stored value — if any — is applied immediately afterwards, before paint
   320→  // of the interaction.)
   321→  useEffect(() => {
   322→    try {
   323→      const stored = window.localStorage.getItem(SIDEBAR_COLLAPSED_STORAGE_KEY);
   324→      if (stored === '1' || stored === '0') {
   325→        const collapsed = stored === '1';
   326→        if (collapsed !== useAppStore.getState().sidebarCollapsed) {
   327→          useAppStore.setState({ sidebarCollapsed: collapsed });

   331→      // storage unavailable — keep expanded default
   332→    }
   333→  }, []);
   334→
   335→  // Check if onboarding should show on first visit (hydration-safe via useSyncExternalStore)
   336→  const onboardingCompleted = useSyncExternalStore(
   337→    (callback) => {
   338→      window.addEventListener('storage', callback);
   339→      return () => window.removeEventListener('storage', callback);
   340→    },
   341→    () => localStorage.getItem('acquisitionos_onboarding_completed') !== null,
   342→    () => true // Server: assume completed (don't show onboarding during SSR)
   343→  );
   344→  // DB-first completion check: the database is the source of truth for
   345→  // onboarding state. A user who already completed (or skipped) the wizard on
   346→  // another device/browser must not be forced through it again just because
   347→  // this browser's localStorage is empty. localStorage is only used to skip
   348→  // the network check instantly when it already knows the answer.
   349→  const [onboardingChecked, setOnboardingChecked] = useState(false);
   350→  useEffect(() => {
   351→    if (onboardingChecked) return;
   352→    if (onboardingCompleted) {
   353→      setOnboardingChecked(true);
   354→      return;

   361→          const data = await res.json();

   363→            data?.settings?.onboardingCompleted === true || data?.progress?.completed === true;
   364→          if (done) {
   365→            // Mirror the DB state into localStorage so subsequent visits skip
   366→            // the network check.
   367→            localStorage.setItem('acquisitionos_onboarding_completed', 'true');
   368→            if (!cancelled) {
   369→              setOnboardingChecked(true);
   370→              return;
   371→            }
   372→          }
   373→        }
   374→      } catch {
   375→        // Network/DB unavailable — fall through to the legacy behavior below.
   376→      }
   377→      if (!cancelled && !onboardingChecked) {
   378→        setOnboardingChecked(true);
   379→        setOnboardingOpen(true);
   380→      }

   382→    return () => {
   383→      cancelled = true;
