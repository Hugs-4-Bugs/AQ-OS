    50→    50→  success: boolean;

    57→    57→// ===== MAIN EXECUTION ENTRY POINT =====
    58→    58→
    59→    59→/**
    60→    60→ * Execute a workflow. Creates a WorkflowExecution record and processes steps.
    61→    61→ * Returns the execution ID immediately (202 Accepted pattern).
    62→    62→ */
    63→    63→export async function executeWorkflow(
    64→    64→  workflowId: string,
    65→    65→  userId: string,
    66→    66→  triggerData?: Record<string, unknown>,
    67→    67→  idempotencyKey?: string
    68→    68→): Promise<ExecuteWorkflowResult> {
    69→    69→  try {
    70→    70→    // Load workflow definition — allow both 'draft' (for testing) and 'active' workflows.
    71→    71→    // Drafts can be executed to validate the workflow before publishing.
    72→    72→    const workflow = await db.workflowDefinition.findFirst({
    73→    73→      where: { id: workflowId, userId, status: { in: ['draft', 'active'] } },
    74→    74→      include: { workflowSteps: { orderBy: { order: 'asc' } } },
    75→    75→    });
    76→    76→
    77→    77→    if (!workflow) {
    78→    78→      return { executionId: '', status: 'failed', error: 'Workflow not found or not active (drafts are executable for testing)' };
    79→    79→    }
    80→    80→
    81→    81→    // Check execution limit
    82→    82→    const limitCheck = await checkExecutionLimit(userId);
    83→    83→    if (!limitCheck.allowed) {
    84→    84→      return {
    85→    85→        executionId: '',
    86→    86→        status: 'failed',
    87→    87→        error: `Monthly execution limit reached (${limitCheck.current}/${limitCheck.limit})`,
    88→    88→      };
    89→    89→    }
    90→    90→
    91→    91→    // Idempotency check
    92→    92→    if (idempotencyKey) {
    93→    93→      const existing = await db.workflowExecution.findFirst({
    94→    94→        where: { idempotencyKey, workflowId },
    95→    95→      });
    96→    96→      if (existing) {
    97→    97→        return { executionId: existing.id, status: existing.status };
    98→    98→      }
    99→    99→    }
   100→   100→
   101→   101→    // Parse steps from nodes/edges if no explicit steps
   102→   102→    let steps: StepDefinition[] = [];
   103→   103→    if (workflow.workflowSteps.length > 0) {
   104→   104→      steps = workflow.workflowSteps.map((s) => ({
   105→   105→        id: s.id,
   106→   106→        type: s.type,
   107→   107→        name: s.name,
   108→   108→        config: JSON.parse(s.config || '{}'),
   109→   109→        order: s.order,
   110→   110→        nextStepId: s.nextStepId || undefined,
   111→   111→      }));
   112→   112→    } else {
   113→   113→      // Parse nodes from JSON
   114→   114→      try {
   115→   115→        const nodes = JSON.parse(workflow.nodes || '[]');
   116→   116→        steps = nodes
   117→   117→          .filter((n: Record<string, unknown>) => n.type !== 'trigger')
   118→   118→          .map((n: Record<string, unknown>, i: number) => ({
   119→   119→            id: String(n.id || `step-${i}`),
   120→   120→            type: String(n.type || 'action'),
   121→   121→            name: String(n.title || n.name || `Step ${i + 1}`),
   122→   122→            config: (n.config as Record<string, unknown>) || {},
   123→   123→            order: i,
   124→   124→          }));
   125→   125→      } catch {
   126→   126→        steps = [];
   127→   127→      }
   128→   128→    }
   129→   129→
   130→   130→    // Create execution record
   131→   131→    const maxRetries = Number(process.env.WORKFLOW_MAX_RETRIES) || 3;
   132→   132→    const timeoutMs = Number(process.env.WORKFLOW_TIMEOUT) || 300000;
   133→   133→
   134→   134→    const execution = await db.workflowExecution.create({
   135→   135→      data: {
   136→   136→        workflowId,
   137→   137→        // Persist the execution owner. The trusted execution context (userId +
   138→   138→        // leadId) must survive on the execution record so that retries,
   139→   139→        // resumes, and audit trails all resolve leads against the SAME user
   140→   140→        // who started the run — never against a null/system owner.
   141→   141→        userId,
   142→   142→        status: 'running',
   143→   143→        triggerData: triggerData ? JSON.stringify(triggerData) : null,
   144→   144→        triggerEvent: triggerData ? JSON.stringify(triggerData) : null,
   145→   145→        currentStep: 0,
   146→   146→        totalSteps: steps.length,
   147→   147→        idempotencyKey: idempotencyKey || null,
   148→   148→        maxRetries,
   149→   149→        timeoutMs,
   150→   150→      },
   151→   151→    });
   152→   152→
   153→   153→    // Execute steps (async but we process them sequentially)
   154→   154→    // We do NOT await the full execution to return 202 quickly
   155→   155→    // But in this single-process model, we run inline
   156→   156→    try {
   157→   157→      await runSteps(execution.id, steps, userId, workflowId, triggerData);
   158→   158→    } catch (error) {
   159→   159→      console.error('[WorkflowEngine] Step execution error:', error);
   160→   160→    }
   161→   161→
   162→   162→    // Log audit event
   163→   163→    await logWorkflowEvent(userId, 'workflow_executed', {
   164→   164→      workflowId,
   165→   165→      executionId: execution.id,
   166→   166→      triggerType: workflow.triggerType,
   167→   167→    });
   168→   168→
   169→   169→    // Emit realtime event
   170→   170→    await emitWorkflowEvent('workflow_started', userId, {
   171→   171→      workflowId,
   172→   172→      executionId: execution.id,
   173→   173→      workflowName: workflow.name,
   174→   174→      triggerType: workflow.triggerType,
   175→   175→    });
   176→   176→
   177→   177→    // Fetch final state
   178→   178→    const finalExecution = await db.workflowExecution.findUnique({
   179→   179→      where: { id: execution.id },
   180→   180→      select: { id: true, status: true, error: true },
   181→   181→    });
   182→   182→
   183→   183→    return {
   184→   184→      executionId: execution.id,
   185→   185→      status: finalExecution?.status || 'running',
   186→   186→      error: finalExecution?.error || undefined,
   187→   187→    };
   188→   188→  } catch (error) {
   189→   189→    console.error('[WorkflowEngine] Failed to execute workflow:', error);
   190→   190→    return {
   191→   191→      executionId: '',
   192→   192→      status: 'failed',
   193→   193→      error: error instanceof Error ? error.message : 'Execution failed',
   194→   194→    };
   195→   195→  }
   196→   196→}
   197→   197→
   198→   198→// ===== STEP RUNNER =====
   199→   199→
   200→   200→async function runSteps(
   201→   201→  executionId: string,
   202→   202→  steps: StepDefinition[],
   203→   203→  userId: string,
   204→   204→  workflowId: string,
   205→   205→  triggerData?: Record<string, unknown>
   206→   206→): Promise<void> {
   207→   207→  const previousOutputs: Record<string, unknown> = {};
   208→   208→  let shouldContinue = true;
   209→   209→
   210→   210→  for (let i = 0; i < steps.length && shouldContinue; i++) {
   211→   211→    const step = steps[i];
   212→   212→
   213→   213→    // Update current step
   214→   214→    await db.workflowExecution.update({
   215→   215→      where: { id: executionId },
   216→   216→      data: { currentStep: i },
   217→   217→    });
   218→   218→
   219→   219→    // Process the step
   220→   220→    const result = await processStep(executionId, step, {
   221→   221→      userId,
   222→   222→      workflowId,
   223→   223→      executionId,
   224→   224→      leadId: triggerData?.leadId as string | undefined,
   225→   225→      triggerData,
   226→   226→      previousOutputs,
   227→   227→    });
   228→   228→
   229→   229→    // Store output
   230→   230→    if (result.output) {
   231→   231→      previousOutputs[step.id] = result.output;
   232→   232→    }
   233→   233→
   234→   234→    // Handle delay steps
   235→   235→    if (step.type === 'delay' || step.type === 'wait_delay') {
   236→   236→      const delayMs = result.output?.delayMs as number | undefined;
   237→   237→      if (delayMs && delayMs > 0 && delayMs <= 86400000) {
   238→   238→        // For short delays (< 24h), sleep inline
   239→   239→        // For longer delays, this should be handled by a queue
   240→   240→        await new Promise(resolve => setTimeout(resolve, Math.min(delayMs, 60000)));
   241→   241→      }
   242→   242→    }
   243→   243→
   244→   244→    // Handle conditional branch — may skip subsequent steps
   245→   245→    if (step.type === 'condition' || step.type === 'conditional_branch') {
   246→   246→      const branch = result.output?.branch as string | undefined;
   247→   247→      if (branch === 'false' || branch === 'skip') {
   248→   248→        // Skip the next step if condition is false
   249→   249→        // (Simple linear branch model)
   250→   250→        i++; // Skip next step
   251→   251→      }
   252→   252→    }
   253→   253→
   254→   254→    if (!result.success) {
   255→   255→      // Mark execution as failed
   256→   256→      await db.workflowExecution.update({
   257→   257→        where: { id: executionId },
   258→   258→        data: {
   259→   259→          status: 'failed',
   260→   260→          error: result.error || 'Step execution failed',
   261→   261→          completedAt: new Date(),
   262→   262→        },
   263→   263→      });
   264→   264→
   265→   265→      // User-facing notification (safe message — raw errors stay in the
   266→   266→      // execution record, never in the notification text).
   267→   267→      await createNotificationOnce({
   268→   268→        userId,
   269→   269→        type: 'workflow_failed',
   270→   270→        title: 'Workflow execution failed',
   271→   271→        message: `A workflow run stopped at step "${step.name || 'unknown'}" and could not continue. You can retry it from the Workflows page.`,
   272→   272→        actionUrl: '/business-ai/workflows',
   273→   273→        metadata: { workflowId, executionId, failedStep: step.name },
   274→   274→        dedupeKey: `wfexec:${executionId}:failed`,
   275→   275→      }).catch(() => {
   276→   276→        // Never fail the execution path because of a notification problem
   277→   277→      });
   278→   278→
   279→   279→      await logWorkflowEvent(userId, 'workflow_failed', {
   280→   280→        workflowId,
   281→   281→        executionId,
   282→   282→        failedStep: step.name,
   283→   283→        error: result.error,
   284→   284→      });
   285→   285→
   286→   286→      // Emit realtime event
   287→   287→      await emitWorkflowEvent('workflow_failed', userId, {
   288→   288→        workflowId,
   289→   289→        executionId,
   290→   290→        failedStep: step.name,
   291→   291→        error: result.error,
   292→   292→      });
   293→   293→
   294→   294→      // Check if we should retry
   295→   295→      const execution = await db.workflowExecution.findUnique({
   296→   296→        where: { id: executionId },
   297→   297→        select: { retryCount: true, maxRetries: true },
   298→   298→      });
   299→   299→
   300→   300→      if (execution && execution.retryCount < execution.maxRetries) {
   301→   301→        await db.workflowExecution.update({
   302→   302→          where: { id: executionId },
   303→   303→          data: { retryCount: { increment: 1 } },
   304→   304→        });
   305→   305→      } else {
   306→   306→        // Max retries exceeded — dead letter
   307→   307→        await sendToDeadLetter(executionId, `Max retries exceeded. Last error: ${result.error || 'Unknown'}`);
   308→   308→      }
   309→   309→
   310→   310→      return;
   311→   311→    }
   312→   312→
   313→   313→    shouldContinue = result.shouldContinue;
   314→   314→  }
   315→   315→
   316→   316→  // Mark execution as completed
   317→   317→  if (shouldContinue) {
   318→   318→    await db.workflowExecution.update({
   319→   319→      where: { id: executionId },
   320→   320→      data: {
   321→   321→        status: 'completed',
   322→   322→        completedAt: new Date(),
   323→   323→        currentStep: steps.length,
   324→   324→      },
   325→   325→    });
   326→   326→
   327→   327→    // User-facing notification (deduped per execution — replay/retry of the
   328→   328→    // same execution id will not create a second notification).
   329→   329→    await createNotificationOnce({
   330→   330→      userId,
   331→   331→      type: 'workflow_completed',
   332→   332→      title: 'Workflow completed',
   333→   333→      message: `A workflow run finished successfully — all ${steps.length} step${steps.length === 1 ? '' : 's'} completed.`,
   334→   334→      actionUrl: '/business-ai/workflows',
   335→   335→      metadata: { workflowId, executionId, totalSteps: steps.length },
   336→   336→      dedupeKey: `wfexec:${executionId}:completed`,
   337→   337→    }).catch(() => {
   338→   338→      // Never fail the execution path because of a notification problem
   339→   339→    });
   340→   340→
   341→   341→    // Emit realtime completion event
   342→   342→    await emitWorkflowEvent('workflow_completed', userId, {
   343→   343→      workflowId,
   344→   344→      executionId,
   345→   345→      totalSteps: steps.length,
   346→   346→    });
   347→   347→  }
   348→   348→}

   350→   350→// ===== PROCESS SINGLE STEP =====
   351→   351→
   352→   352→export async function processStep(
   353→   353→  executionId: string,
   354→   354→  step: StepDefinition,
   355→   355→  context: ActionContext
   356→   356→): Promise<ProcessStepResult> {
   357→   357→  const startTime = Date.now();
   358→   358→
   359→   359→  try {
   360→   360→    let result: ActionResult;
   361→   361→
   362→   362→    // Deduct credits before execution
   363→   363→    const creditResult = await deductExecutionCredits(context.userId, step.type);
   364→   364→
   365→   365→    if (!creditResult.success && creditResult.creditsUsed > 0) {
   366→   366→      // Only fail if the action actually costs credits and we can't afford them
   367→   367→      const durationMs = Date.now() - startTime;
   368→   368→
   369→   369→      await db.workflowLog.create({
   370→   370→        data: {
   371→   371→          executionId,
   372→   372→          stepName: step.name,
   373→   373→          stepType: step.type,
   374→   374→          status: 'failed',
   375→   375→          input: JSON.stringify(step.config),
   376→   376→          error: `Insufficient credits: ${creditResult.error}`,
   377→   377→          durationMs,
   378→   378→          userId: context.userId,
   379→   379→        },
   380→   380→      });
   381→   381→
   382→   382→      return {
   383→   383→        success: false,
   384→   384→        error: `Insufficient credits: ${creditResult.error}`,
   385→   385→        shouldContinue: false,
   386→   386→        durationMs,
   387→   387→      };
   388→   388→    }
   389→   389→
   390→   390→    // Execute the action.
   391→   391→    // WorkflowStep.type stores the NODE type ('action' | 'delay' | 'condition' | 'ai_action'),
   392→   392→    // while executeAction dispatches on the ACTION type ('send_email' | 'wait_delay' | ...).
   393→   393→    // The action type always rides in config.actionType; fall back to step.type for
   394→   394→    // legacy steps that stored the action type directly.
   395→   395→    const actionType = String(step.config?.actionType || step.type);
   396→   396→    result = await executeAction(actionType, step.config, context);
   397→   397→
   398→   398→    const durationMs = Date.now() - startTime;
   399→   399→
   400→   400→    // Create workflow log entry
   401→   401→    await db.workflowLog.create({
   402→   402→      data: {
   403→   403→        executionId,
   404→   404→        stepName: step.name,
   405→   405→        stepType: step.type,
   406→   406→        status: result.success ? 'success' : 'failed',
   407→   407→        input: JSON.stringify(step.config),
   408→   408→        output: result.output ? JSON.stringify(result.output) : null,
   409→   409→        error: result.error || null,
   410→   410→        durationMs,
   411→   411→        userId: context.userId,
   412→   412→      },
   413→   413→    });
   414→   414→
   415→   415→    // Emit step event
   416→   416→    await emitWorkflowEvent(
   417→   417→      result.success ? 'workflow_step_completed' : 'workflow_step_failed',
   418→   418→      context.userId,
   419→   419→      {
   420→   420→        executionId,
   421→   421→        stepName: step.name,
   422→   422→        stepType: step.type,
   423→   423→        durationMs,
   424→   424→        error: result.error,
   425→   425→      }
   426→   426→    );
   427→   427→
   428→   428→    return {
   429→   429→      success: result.success,
   430→   430→      output: result.output,
   431→   431→      error: result.error,
   432→   432→      shouldContinue: result.success,
   433→   433→      durationMs,
   434→   434→    };
   435→   435→  } catch (error) {
   436→   436→    const durationMs = Date.now() - startTime;
   437→   437→
   438→   438→    await db.workflowLog.create({
   439→   439→      data: {
   440→   440→        executionId,
   441→   441→        stepName: step.name,
   442→   442→        stepType: step.type,
   443→   443→        status: 'failed',
   444→   444→        input: JSON.stringify(step.config),
   445→   445→        error: error instanceof Error ? error.message : 'Step execution failed',
   446→   446→        durationMs,
   447→   447→        userId: context.userId,
   448→   448→      },
   449→   449→    });
   450→   450→
   451→   451→    return {
   452→   452→      success: false,
   453→   453→      error: error instanceof Error ? error.message : 'Step execution failed',
   454→   454→      shouldContinue: false,
   455→   455→      durationMs,
   456→   456→    };
   457→   457→  }
   458→   458→}

   460→   460→// ===== PAUSE EXECUTION =====
   461→   461→
   462→   462→export async function pauseExecution(
   463→   463→  executionId: string,
   464→   464→  userId: string
   465→   465→): Promise<{ success: boolean; error?: string }> {
   466→   466→  try {
   467→   467→    const execution = await db.workflowExecution.findFirst({
   468→   468→      where: { id: executionId },
   469→   469→      include: { workflow: { select: { userId: true } } },
   470→   470→    });
   471→   471→
   472→   472→    if (!execution) {
   473→   473→      return { success: false, error: 'Execution not found' };
   474→   474→    }
   475→   475→
   476→   476→    if (execution.workflow.userId !== userId) {
   477→   477→      return { success: false, error: 'Not authorized' };
   478→   478→    }
   479→   479→
   480→   480→    if (execution.status !== 'running') {
   481→   481→      return { success: false, error: `Cannot pause execution in '${execution.status}' status` };
   482→   482→    }
   483→   483→
   484→   484→    await db.workflowExecution.update({
   485→   485→      where: { id: executionId },
   486→   486→      data: { status: 'paused', pausedAt: new Date() },
   487→   487→    });
   488→   488→
   489→   489→    await logWorkflowEvent(userId, 'workflow_paused', { executionId });
   490→   490→
   491→   491→    // Emit realtime event
   492→   492→    await emitWorkflowEvent('workflow_paused', userId, { executionId });
   493→   493→
   494→   494→    return { success: true };
   495→   495→  } catch (error) {
   496→   496→    return {
   497→   497→      success: false,
   498→   498→      error: error instanceof Error ? error.message : 'Failed to pause execution',
   499→   499→    };
   500→   500→  }
   501→   501→}

   503→   503→// ===== RESUME EXECUTION =====
   504→   504→
   505→   505→export async function resumeExecution(
   506→   506→  executionId: string,
   507→   507→  userId: string
   508→   508→): Promise<{ success: boolean; error?: string }> {
   509→   509→  try {
   510→   510→    const execution = await db.workflowExecution.findFirst({
   511→   511→      where: { id: executionId },
   512→   512→      include: { workflow: { select: { userId: true } } },
   513→   513→    });
   514→   514→
   515→   515→    if (!execution) {
   516→   516→      return { success: false, error: 'Execution not found' };
   517→   517→    }
   518→   518→
   519→   519→    if (execution.workflow.userId !== userId) {
   520→   520→      return { success: false, error: 'Not authorized' };
   521→   521→    }
   522→   522→
   523→   523→    if (execution.status !== 'paused') {
   524→   524→      return { success: false, error: `Cannot resume execution in '${execution.status}' status` };
   525→   525→    }
   526→   526→
   527→   527→    await db.workflowExecution.update({
   528→   528→      where: { id: executionId },
   529→   529→      data: { status: 'running', resumedAt: new Date() },
   530→   530→    });
   531→   531→
   532→   532→    // Re-run remaining steps
   533→   533→    const workflow = await db.workflowDefinition.findUnique({
   534→   534→      where: { id: execution.workflowId },
   535→   535→      include: { workflowSteps: { orderBy: { order: 'asc' } } },
   536→   536→    });
   537→   537→
   538→   538→    if (workflow) {
   539→   539→      const steps = workflow.workflowSteps.map((s) => ({
   540→   540→        id: s.id,
   541→   541→        type: s.type,
   542→   542→        name: s.name,
   543→   543→        config: JSON.parse(s.config || '{}'),
   544→   544→        order: s.order,
   545→   545→        nextStepId: s.nextStepId || undefined,
   546→   546→      }));
   547→   547→
   548→   548→      const triggerData = execution.triggerData ? JSON.parse(execution.triggerData) : undefined;
   549→   549→      const remainingSteps = steps.slice(execution.currentStep);
   550→   550→
   551→   551→      await runSteps(executionId, remainingSteps, userId, execution.workflowId, triggerData);
   552→   552→    }
   553→   553→
   554→   554→    await logWorkflowEvent(userId, 'workflow_resumed', { executionId });
   555→   555→
   556→   556→    // Emit realtime event
   557→   557→    await emitWorkflowEvent('workflow_resumed', userId, { executionId });
   558→   558→
   559→   559→    return { success: true };
   560→   560→  } catch (error) {
   561→   561→    return {
   562→   562→      success: false,
   563→   563→      error: error instanceof Error ? error.message : 'Failed to resume execution',
   564→   564→    };
   565→   565→  }
   566→   566→}

   613→   613→export async function retryExecution(
   614→   614→  executionId: string,
   615→   615→  userId: string
   616→   616→): Promise<{ success: boolean; executionId?: string; error?: string }> {
   617→   617→  try {
   618→   618→    const execution = await db.workflowExecution.findFirst({
   619→   619→      where: { id: executionId },
   620→   620→      include: { workflow: { select: { userId: true, id: true } } },
   621→   621→    });
   622→   622→
   623→   623→    if (!execution) {
   624→   624→      return { success: false, error: 'Execution not found' };
   625→   625→    }
   626→   626→
   627→   627→    if (execution.workflow.userId !== userId) {
   628→   628→      return { success: false, error: 'Not authorized' };
   629→   629→    }
   630→   630→
   631→   631→    if (!['failed', 'cancelled'].includes(execution.status)) {
   632→   632→      return { success: false, error: `Cannot retry execution in '${execution.status}' status` };
   633→   633→    }
   634→   634→
   635→   635→    // Re-execute from the failed step
   636→   636→    const workflow = await db.workflowDefinition.findUnique({
   637→   637→      where: { id: execution.workflowId },
   638→   638→      include: { workflowSteps: { orderBy: { order: 'asc' } } },
   639→   639→    });
   640→   640→
   641→   641→    if (!workflow) {
   642→   642→      return { success: false, error: 'Workflow not found' };
   643→   643→    }
   644→   644→
   645→   645→    const steps = workflow.workflowSteps.map((s) => ({
   646→   646→      id: s.id,
   647→   647→      type: s.type,
   648→   648→      name: s.name,
   649→   649→      config: JSON.parse(s.config || '{}'),
   650→   650→      order: s.order,
   651→   651→      nextStepId: s.nextStepId || undefined,
   652→   652→    }));
   653→   653→
   654→   654→    // Reset execution state
   655→   655→    await db.workflowExecution.update({
   656→   656→      where: { id: executionId },
   657→   657→      data: {
   658→   658→        status: 'running',
   659→   659→        error: null,
   660→   660→        completedAt: null,
   661→   661→        retryCount: { increment: 1 },
   662→   662→      },
   663→   663→    });
   664→   664→
   665→   665→    const triggerData = execution.triggerData ? JSON.parse(execution.triggerData) : undefined;
   666→   666→    const remainingSteps = steps.slice(execution.currentStep);
   667→   667→
   668→   668→    await runSteps(executionId, remainingSteps, userId, execution.workflowId, triggerData);
   669→   669→
   670→   670→    await logWorkflowEvent(userId, 'workflow_retried', {
   671→   671→      executionId,
   672→   672→      workflowId: execution.workflowId,
   673→   673→    });
   674→   674→
   675→   675→    // Emit realtime event
   676→   676→    await emitWorkflowEvent('workflow_retried', userId, {
   677→   677→      executionId,
   678→   678→      workflowId: execution.workflowId,
   679→   679→    });
   680→   680→
   681→   681→    return { success: true, executionId };
   682→   682→  } catch (error) {
   683→   683→    return {
   684→   684→      success: false,
   685→   685→      error: error instanceof Error ? error.message : 'Failed to retry execution',
   686→   686→    };
   687→   687→  }
   688→   688→}

   690→   690→// ===== EXECUTION HISTORY =====
   691→   691→
   692→   692→export async function getExecutionHistory(
   693→   693→  userId: string,
   694→   694→  filters?: {
   695→   695→    status?: string;
   696→   696→    workflowId?: string;
   697→   697→    dateFrom?: string;
   698→   698→    dateTo?: string;
   699→   699→    page?: number;
   700→   700→    limit?: number;
   701→   701→  }
   702→   702→): Promise<{ executions: Record<string, unknown>[]; total: number; page: number; totalPages: number }> {
   703→   703→  try {
   704→   704→    const page = filters?.page || 1;
   705→   705→    const limit = Math.min(filters?.limit || 20, 100);
   706→   706→    const offset = (page - 1) * limit;
   707→   707→
   708→   708→    const where: Record<string, unknown> = {
   709→   709→      workflow: { userId },
   710→   710→    };
   711→   711→
   712→   712→    if (filters?.status) where.status = filters.status;
   713→   713→    if (filters?.workflowId) where.workflowId = filters.workflowId;
   714→   714→    if (filters?.dateFrom || filters?.dateTo) {
   715→   715→      const startedAt: Record<string, Date> = {};
   716→   716→      if (filters?.dateFrom) startedAt.gte = new Date(filters.dateFrom);
   717→   717→      if (filters?.dateTo) startedAt.lte = new Date(filters.dateTo);
   718→   718→      where.startedAt = startedAt;
   719→   719→    }
   720→   720→
   721→   721→    const [executions, total] = await Promise.all([
   722→   722→      db.workflowExecution.findMany({
   723→   723→        where,
   724→   724→        orderBy: { startedAt: 'desc' },
   725→   725→        skip: offset,
   726→   726→        take: limit,
   727→   727→        include: {
   728→   728→          workflow: { select: { name: true, triggerType: true } },
   729→   729→        },
   730→   730→      }),
   731→   731→      db.workflowExecution.count({ where }),
   732→   732→    ]);
   733→   733→
   734→   734→    return {
   735→   735→      executions: executions.map((e) => ({
   736→   736→        id: e.id,
   737→   737→        workflowId: e.workflowId,
   738→   738→        workflowName: e.workflow.name,
   739→   739→        triggerType: e.workflow.triggerType,
   740→   740→        status: e.status,
   741→   741→        currentStep: e.currentStep,
   742→   742→        totalSteps: e.totalSteps,
   743→   743→        startedAt: e.startedAt,
   744→   744→        completedAt: e.completedAt,
   745→   745→        error: e.error,
   746→   746→        retryCount: e.retryCount,
   747→   747→        deadLettered: e.deadLettered,
   748→   748→        deadLetterReason: e.deadLetterReason,
   749→   749→      })),
   750→   750→      total,
   751→   751→      page,
   752→   752→      totalPages: Math.ceil(total / limit),
   753→   753→    };
   754→   754→  } catch (error) {
   755→   755→    console.error('[WorkflowEngine] Failed to get execution history:', error);
   756→   756→    return { executions: [], total: 0, page: 1, totalPages: 0 };
   757→   757→  }
   758→   758→}

   819→   819→
   820→   820→export async function getExecutionMetrics(
   821→   821→  userId: string
   822→   822→): Promise<Record<string, unknown>> {
   823→   823→  try {
   824→   824→    // Total executions
   825→   825→    const total = await db.workflowExecution.count({
   826→   826→      where: { workflow: { userId } },
   827→   827→    });
   828→   828→
   829→   829→    // Success count
   830→   830→    const completed = await db.workflowExecution.count({
   831→   831→      where: { workflow: { userId }, status: 'completed' },
   832→   832→    });
   833→   833→
   834→   834→    // Failed count
   835→   835→    const failed = await db.workflowExecution.count({
   836→   836→      where: { workflow: { userId }, status: 'failed' },
   837→   837→    });
   838→   838→
   839→   839→    // Running count
   840→   840→    const running = await db.workflowExecution.count({
   841→   841→      where: { workflow: { userId }, status: 'running' },
   842→   842→    });
   843→   843→
   844→   844→    // Dead lettered count
   845→   845→    const deadLettered = await db.workflowExecution.count({
   846→   846→      where: { workflow: { userId }, deadLettered: true },
   847→   847→    });
   848→   848→
   849→   849→    // Success rate
   850→   850→    const successRate = total > 0 ? ((completed / total) * 100).toFixed(1) : '0';
   851→   851→
   852→   852→    // Average duration (for completed executions)
   853→   853→    const completedExecutions = await db.workflowExecution.findMany({
   854→   854→      where: { workflow: { userId }, status: 'completed', completedAt: { not: null } },
   855→   855→      select: { startedAt: true, completedAt: true },
   856→   856→      take: 100,
   857→   857→    });
   858→   858→
   859→   859→    const durations = completedExecutions
   860→   860→      .filter((e) => e.completedAt)
   861→   861→      .map((e) => e.completedAt!.getTime() - e.startedAt.getTime());
   862→   862→
   863→   863→    const avgDurationMs = durations.length > 0
   864→   864→      ? Math.round(durations.reduce((a, b) => a + b, 0) / durations.length)
   865→   865→      : 0;
   866→   866→
   867→   867→    // By trigger type
   868→   868→    const byTriggerType = await db.workflowExecution.groupBy({
   869→   869→      by: ['workflowId'],
   870→   870→      where: { workflow: { userId } },
   871→   871→      _count: { id: true },
   872→   872→    });
   873→   873→
   874→   874→    // By status
   875→   875→    const byStatus = await db.workflowExecution.groupBy({
   876→   876→      by: ['status'],
   877→   877→      where: { workflow: { userId } },
   878→   878→      _count: { id: true },
   879→   879→    });
   880→   880→
   881→   881→    // This month
   882→   882→    const startOfMonth = new Date(new Date().getFullYear(), new Date().getMonth(), 1);
   883→   883→    const thisMonth = await db.workflowExecution.count({
   884→   884→      where: { workflow: { userId }, startedAt: { gte: startOfMonth } },
   885→   885→    });
   886→   886→
   887→   887→    return {
   888→   888→      total,
   889→   889→      completed,
   890→   890→      failed,
   891→   891→      running,
   892→   892→      deadLettered,
   893→   893→      successRate: `${successRate}%`,
   894→   894→      avgDurationMs,
   895→   895→      avgDurationFormatted: formatDuration(avgDurationMs),
   896→   896→      thisMonth,
   897→   897→      byStatus: byStatus.map((s) => ({ status: s.status, count: s._count.id })),
   898→   898→      byTriggerType: byTriggerType.map((t) => ({ workflowId: t.workflowId, count: t._count.id })),
   899→   899→    };
   900→   900→  } catch (error) {
   901→   901→    console.error('[WorkflowEngine] Failed to get execution metrics:', error);
   902→   902→    return {
   903→   903→      total: 0,
   904→   904→      completed: 0,
   905→   905→      failed: 0,
   906→   906→      running: 0,
   907→   907→      deadLettered: 0,
   908→   908→      successRate: '0%',
   909→   909→      avgDurationMs: 0,
   910→   910→      thisMonth: 0,
   911→   911→    };
   912→   912→  }
   913→   913→}
   914→   914→
   915→   915→function formatDuration(ms: number): string {
   916→   916→  if (ms < 1000) return `${ms}ms`;
   917→   917→  if (ms < 60000) return `${(ms / 1000).toFixed(1)}s`;
   918→   918→  if (ms < 3600000) return `${(ms / 60000).toFixed(1)}m`;
   919→   919→  return `${(ms / 3600000).toFixed(1)}h`;
   920→   920→}
