     1→/**
     2→ * AcquisitionOS Email Service
     3→ *
     4→ * Production-grade email service with provider fallback chain:
     5→ *   1. Resend (primary — modern email API, real key required)
     6→ *   2. Nodemailer SMTP (fallback — self-hosted / custom SMTP, real creds required)
     7→ *   3. Ethereal Email (development/testing — REAL SMTP to a test mailbox
     8→ *      with a browser-previewable URL; not simulation)
     9→ *   4. Console logger (last resort — AUTH_BYPASS_EMAIL=true only)
    10→ *
    11→ * All errors are caught and logged — this service never throws in a way that
    12→ * would block the main auth flow.
    13→ */
    14→

    26→export interface EmailResult {

    28→  messageId?: string;
    29→  error?: string;

    42→  provider?: 'resend' | 'smtp' | 'ethereal' | 'console';

    68→/**
    69→ * Returns `true` when at least one real email provider is configured.
    70→ *
    71→ * "Configured" means either:
    72→ *   - A real Resend API key (not a placeholder like `re_your-...`), OR
    73→ *   - Real SMTP credentials (not `your-email@gmail.com` / `your-app-password`), OR
    74→ *   - We are in Ethereal mode (which is REAL SMTP delivery to a test mailbox).
    75→ *
    76→ * Auth routes use this to decide whether to actually attempt sending. In
    77→ * Ethereal mode this always returns `true` so that the magic-link / OTP /
    78→ * signup / reset flows perform real email delivery and surface the preview URL.
    79→ */
    80→export function isEmailServiceConfigured(): boolean {
    81→  return isRealResendConfigured() || isRealSmtpConfigured() || isEtherealMode();
    82→}

    84→/**
    85→ * Returns the human-readable name of the provider that *would* be used right
    86→ * now. Useful for logging and for the `emailProvider` field in API responses.
    87→ */
    88→export function activeEmailProvider(): 'resend' | 'smtp' | 'ethereal' | 'console' {
    89→  if (isRealResendConfigured()) return 'resend';
    90→  if (isRealSmtpConfigured()) return 'smtp';
    91→  if (isEtherealMode()) return 'ethereal';
    92→  return 'console';
    93→}

    99→async function sendViaResend(
   100→  payload: EmailPayload,
   101→  from: string,
   102→): Promise<EmailResult> {
   103→  try {
   104→    // Dynamic import so the app doesn't crash if `resend` is not installed.
   105→    const { Resend } = await import("resend");
   106→    const resend = new Resend(process.env.RESEND_API_KEY!);
   107→
   108→    // Build email params — include attachments if provided
   109→    const emailParams: Record<string, unknown> = {
   110→      from,
   111→      to: payload.to,
   112→      subject: payload.subject,
   113→      html: payload.html,
   114→      text: payload.text,
   115→    };
   116→
   117→    if (payload.attachments && payload.attachments.length > 0) {
   118→      emailParams.attachments = payload.attachments.map((att) => ({
   119→        filename: att.filename,
   120→        content:
   121→          typeof att.content === "string"
   122→            ? att.content
   123→            : att.content.toString("base64"),
   124→        ...(att.contentType ? { content_type: att.contentType } : {}),
   125→      }));
   126→    }
   127→
   128→    const { data, error } = await resend.emails.send(emailParams as Parameters<typeof resend.emails.send>[0]);
   129→
   130→    if (error) {
   131→      console.error("[EmailService] Resend error:", error);
   132→      return { sent: false, error: error.message };
   133→    }
   134→
   135→    return { sent: true, messageId: data?.id, provider: 'resend' };
   136→  } catch (err: unknown) {
   137→    const message =
   138→      err instanceof Error ? err.message : "Failed to send via Resend";
   139→    console.error("[EmailService] Resend exception:", message);
   140→    return { sent: false, error: message };
   141→  }
   142→}

   147→
   148→async function sendViaSmtp(
   149→  payload: EmailPayload,
   150→): Promise<EmailResult> {
   151→  try {
   152→    // nodemailer is a project dependency — import statically is fine, but we
   153→    // keep the dynamic form for symmetry with the Resend path.
   154→    const nodemailer = await import("nodemailer");
   155→
   156→    const smtpConfig = {
   157→      host: process.env.SMTP_HOST,
   158→      port: Number(process.env.SMTP_PORT) || 587,
   159→      secure: Number(process.env.SMTP_PORT) === 465,
   160→      auth: {
   161→        user: process.env.SMTP_USER,
   162→        pass: getSmtpPassword(),
   163→      },
   164→    };
   165→
   166→    console.log(`[EmailService] SMTP connecting to ${smtpConfig.host}:${smtpConfig.port} (secure=${smtpConfig.secure}, user=${smtpConfig.auth.user ? 'SET' : 'MISSING'})`);
   167→
   168→    const transporter = nodemailer.createTransport(smtpConfig);
   169→
   170→    const fromAddress =
   171→      process.env.SMTP_FROM || "AcquisitionOS <noreply@acquisitionos.com>";
   172→
   173→    const mailOptions: Record<string, unknown> = {
   174→      from: fromAddress,
   175→      to: payload.to,
   176→      subject: payload.subject,
   177→      html: payload.html,
   178→      text: payload.text,
   179→    };
   180→
   181→    // Nodemailer supports attachments[] natively
   182→    if (payload.attachments && payload.attachments.length > 0) {
   183→      mailOptions.attachments = payload.attachments.map((att) => ({
   184→        filename: att.filename,
   185→        content: att.content,
   186→        ...(att.contentType ? { contentType: att.contentType } : {}),
   187→      }));
   188→    }
   189→
   190→    const info = await transporter.sendMail(mailOptions);
   191→
   192→    console.log(`[EmailService] ✓ SMTP mail sent: messageId=${info.messageId}, response=${info.response}`);
   193→    return { sent: true, messageId: info.messageId, provider: 'smtp' };
   194→  } catch (err: unknown) {
   195→    const message =
   196→      err instanceof Error ? err.message : "Failed to send via SMTP";
   197→    console.error("[EmailService] SMTP exception:", message);
   198→    return { sent: false, error: message };
   199→  }
   200→}

   202→// ---------------------------------------------------------------------------
   203→// Provider: Ethereal Email (REAL SMTP to test mailbox)
   204→// ---------------------------------------------------------------------------
   205→
   206→/**
   207→ * Send via Ethereal Email — a real SMTP test service operated by the
   208→ * Nodemailer team. The message is delivered over the real SMTP protocol to a
   209→ * real (private) mailbox, and `nodemailer.getTestMessageUrl(info)` returns a
   210→ * permanent preview URL where the rendered email can be viewed in a browser.
   211→ *
   212→ * This is NOT simulation: it exercises the exact same SMTP code path as
   213→ * production SMTP delivery, just against ethereal.email instead of Gmail /
   214→ * SendGrid / etc. The preview URL is returned so auth routes can surface it
   215→ * to the UI for the user to click the magic link / read the OTP.
   216→ */
   217→async function sendViaEthereal(
   218→  payload: EmailPayload,
   219→  from: string,
   220→): Promise<EmailResult> {
   221→  try {
   222→    const nodemailer = await import("nodemailer");
   223→    const transport = await getEtherealTransport();
   224→
   225→    const mailOptions: Record<string, unknown> = {
   226→      from,
   227→      to: payload.to,
   228→      subject: payload.subject,
   229→      html: payload.html,
   230→      text: payload.text,
   231→    };
   232→
   233→    if (payload.attachments && payload.attachments.length > 0) {
   234→      mailOptions.attachments = payload.attachments.map((att) => ({
   235→        filename: att.filename,
   236→        content: att.content,
   237→        ...(att.contentType ? { contentType: att.contentType } : {}),
   238→      }));
   239→    }
   240→
   241→    const info = await transport.sendMail(mailOptions);
   242→    const previewUrl =
   243→      (nodemailer.getTestMessageUrl(info) as string | undefined) || undefined;
   244→
   245→    console.log(
   246→      `[EmailService] ✓ Ethereal mail sent to ${payload.to} (messageId=${info.messageId})`,
   247→    );
   248→    if (previewUrl) {
   249→      console.log(`[EmailService]   Preview URL: ${previewUrl}`);
   250→    }
   251→
   252→    return {
   253→      sent: true,
   254→      messageId: info.messageId,
   255→      previewUrl,
   256→      provider: 'ethereal',
   257→    };
   258→  } catch (err: unknown) {
   259→    const message =
   260→      err instanceof Error ? err.message : "Failed to send via Ethereal";
   261→    console.error("[EmailService] Ethereal exception:", message);
   262→    return { sent: false, error: message };
   263→  }
   264→}

   284→// ---------------------------------------------------------------------------
   285→// Core send function (fallback chain)
   286→// ---------------------------------------------------------------------------
   287→
   288→export async function sendEmail(payload: EmailPayload): Promise<EmailResult> {
   289→  const provider = activeEmailProvider();
   290→  console.log(
   291→    `[EmailService] sendEmail called: to=${payload.to}, subject=${payload.subject}, activeProvider=${provider}`,
   292→  );
   293→
   294→  // 1. Email bypass override — log to console instead of sending.
   295→  // This uses the centralized feature-flags system (AUTH_BYPASS_EMAIL / AUTH_DEV_MODE).
   296→  // Only triggered when explicitly opted in; never the default.
   297→  if (featureFlagBypass()) {
   298→    console.log("[EmailService] AUTH_BYPASS_EMAIL=true — logging instead of sending");
   299→    return sendViaConsole(payload);
   300→  }
   301→
   302→  const from =
   303→    process.env.EMAIL_FROM || process.env.SMTP_FROM || "AcquisitionOS <noreply@acquisitionos.com>";
   304→
   305→  // 2. Try Resend (primary) — only when a REAL API key is configured.
   306→  // Placeholder values like `re_your-...` are skipped to avoid a guaranteed 401.
   307→  if (isRealResendConfigured()) {
   308→    console.log("[EmailService] Attempting Resend…");
   309→    const result = await sendViaResend(payload, from);
   310→    if (result.sent) {
   311→      console.log(
   312→        `[EmailService] ✓ Sent via Resend to ${payload.to} (id: ${result.messageId})`,
   313→      );
   314→      return result;
   315→    }
   316→    console.warn("[EmailService] ✗ Resend failed, falling back…", result.error);
   317→  }
   318→
   319→  // 3. Try REAL SMTP — only when real credentials are configured.
   320→  // Placeholders (`your-email@gmail.com`, `your-app-password`) are skipped so
   321→  // we don't waste cycles failing Gmail auth and triggering dev fallbacks.
   322→  if (isRealSmtpConfigured()) {
   323→    console.log(
   324→      `[EmailService] Attempting SMTP to ${process.env.SMTP_HOST}:${process.env.SMTP_PORT}…`,
   325→    );
   326→    const result = await sendViaSmtp(payload);
   327→    if (result.sent) {
   328→      console.log(
   329→        `[EmailService] ✓ Sent via SMTP to ${payload.to} (id: ${result.messageId})`,
   330→      );
   331→      return result;
   332→    }
   333→    console.warn("[EmailService] ✗ SMTP failed, falling back…", result.error);
   334→  }
   335→
   336→  // 4. Ethereal Email — REAL SMTP to a test mailbox with a preview URL.
   337→  // This is the development/testing fallback that makes the entire auth flow
   338→  // (magic link, OTP, signup verification, password reset) work end-to-end
   339→  // without any mock code or token-in-response shortcuts.
   340→  if (isEtherealMode()) {
   341→    console.log("[EmailService] Attempting Ethereal (real SMTP to test inbox)…");
   342→    return await sendViaEthereal(payload, from);
   343→  }
   344→
   345→  // 5. Last resort — console log. This branch is only reached if a real
   346→  // provider was configured but explicitly skipped above (shouldn't happen
   347→  // in practice) and Ethereal is unavailable for some reason.
   348→  console.warn(
   349→    "[EmailService] ⚠ No email provider available — logging to console",
   350→  );
   351→  return sendViaConsole(payload);
   352→}

   482→
   483→function escapeHtml(str: string): string {
   484→  return str
   485→    .replace(/&/g, "&amp;")
   486→    .replace(/</g, "&lt;")
   487→    .replace(/>/g, "&gt;")
   488→    .replace(/"/g, "&quot;")
   489→    .replace(/'/g, "&#039;");
   490→}

   497→ * Email verification OTP — sent when a user signs up or changes their email.

   499→export async function sendVerificationEmail(
   500→  email: string,
   501→  name: string,
   502→  otp: string,
   503→): Promise<EmailResult> {
   504→  const subject = "Verify your email — AcquisitionOS";

   512→    ${otpBlock(otp)}
   513→    <p style="margin:0; font-size:13px; line-height:20px; color:${TEXT_SECONDARY};">
   514→      This code expires in 10 minutes.
   515→    </p>
   516→    ${securityNote()}
   517→    ${signOff()}`,
   518→    preview,
   519→  );

   521→  const text = `Hi ${name},\n\nWelcome to AcquisitionOS! Your verification code is: ${otp}\n\nThis code expires in 10 minutes.\n\nIf you didn't request this, you can safely ignore this email.\n\n— The AcquisitionOS Team`;
   522→
   523→  return sendEmail({ to: email, subject, html, text });
   524→}
