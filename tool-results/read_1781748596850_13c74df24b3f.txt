    46→/** Password strength calculator */
    47→function getPasswordStrength(password: string): {
    48→  score: number; // 0–4
    49→  label: string;
    50→  color: string;
    51→} {
    52→  if (!password) return { score: 0, label: '', color: '' };
    53→
    54→  let score = 0;
    55→  if (password.length >= 8) score++;
    56→  if (password.length >= 12) score++;
    57→  if (/[A-Z]/.test(password) && /[a-z]/.test(password)) score++;
    58→  if (/\d/.test(password)) score++;
    59→  if (/[^A-Za-z0-9]/.test(password)) score++;
    60→
    61→  // Normalize to 0–4
    62→  score = Math.min(4, score);
    63→
    64→  const labels = ['', 'Weak', 'Fair', 'Good', 'Strong'];
    65→  const colors = [
    66→    '',
    67→    'oklch(0.6 0.25 15)',   // red
    68→    'oklch(0.75 0.18 55)',  // amber
    69→    'oklch(0.85 0.18 90)',  // yellow-green
    70→    'oklch(0.72 0.19 155)', // green
    71→  ];
    72→
    73→  return { score, label: labels[score], color: colors[score] };
    74→}
    75→
    76→/** Email validation */
    77→function isValidEmail(email: string): boolean {
    78→  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
    79→}

   238→function PasswordStrengthIndicator({ password }: { password: string }) {
   239→  const strength = getPasswordStrength(password);
   240→  if (!password) return null;
   241→
   242→  const percent = (strength.score / 4) * 100;
   243→
   244→  return (
   245→    <div className="space-y-1.5">
   246→      <Progress
   247→        value={percent}
   248→        className="h-1.5"
   249→        style={
   250→          {
   251→            '--progress-color': strength.color,
   252→          } as React.CSSProperties
   253→        }
   254→      />
   255→      <p className="text-xs" style={{ color: strength.color }}>
   256→        {strength.label}
   257→      </p>
   258→    </div>
   259→  );
   260→}

   298→export function SignInPage({

   301→  onSuccess,
   302→  onMagicLinkClick,
   303→  onOtpLoginClick,
   304→}: {

   306→  onForgotPasswordClick?: () => void;
   307→  onSuccess?: () => void;
   308→  onMagicLinkClick?: () => void;
   309→  onOtpLoginClick?: () => void;
   310→}) {
   311→  const { signIn, isLoading } = useAuth();
   312→  const [email, setEmail] = useState('');
   313→  const [password, setPassword] = useState('');
   314→  const [showPassword, setShowPassword] = useState(false);
   315→  const [errors, setErrors] = useState<Record<string, string>>({});
   316→  const [generalError, setGeneralError] = useState('');
   317→  const [authConfig, setAuthConfig] = useState<{ googleAvailable: boolean; loaded: boolean }>({ googleAvailable: false, loaded: false });
   318→
   319→  // Fetch auth config on mount to check Google OAuth availability
   320→  useEffect(() => {
   321→    fetch('/api/auth/config')
   322→      .then(res => res.ok ? res.json() : { googleAvailable: false })
   323→      .then(data => setAuthConfig({ googleAvailable: !!data.googleAvailable, loaded: true }))
   324→      .catch(() => setAuthConfig({ googleAvailable: false, loaded: true }));
   325→  }, []);
   326→
   327→  // Server-driven Google OAuth: client_id never touches frontend
   328→  const startGoogleOAuth = useCallback(async () => {
   329→    try {
   330→      const origin = typeof window !== 'undefined' ? window.location.origin : '';
   331→      const res = await fetch(`/api/auth/google/state?origin=${encodeURIComponent(origin)}`);
   332→      if (!res.ok) throw new Error('Failed to get OAuth state');
   333→      const data = await res.json();
   334→      if (data.authUrl) {
   335→        window.location.href = data.authUrl;
   336→      }
   337→    } catch (err) {
   338→      console.error('Google OAuth error:', err);
   339→      toast.error('Google sign-in is not available in this environment.');
   340→    }
   341→  }, []);
   342→
   343→  const validate = useCallback(() => {
   344→    const errs: Record<string, string> = {};
   345→    if (!email) errs.email = 'Email is required';
   346→    else if (!isValidEmail(email)) errs.email = 'Enter a valid email address';
   347→    if (!password) errs.password = 'Password is required';
   348→    else if (password.length < 6) errs.password = 'Password must be at least 6 characters';
   349→    return errs;
   350→  }, [email, password]);
   351→
   352→  const handleSubmit = useCallback(
   353→    async (e: React.FormEvent) => {
   354→      e.preventDefault();
   355→      setGeneralError('');
   356→      const validationErrors = validate();
   357→      setErrors(validationErrors);
   358→      if (Object.keys(validationErrors).length > 0) return;
   359→
   360→      const result = await signIn(email, password);
   361→      if (result.success) {
   362→        toast.success('Welcome back!');
   363→        onSuccess?.();
   364→      } else if (result.mfaRequired) {
   365→        toast.info('MFA verification required');
   366→        // Caller should handle MFA redirect via mfaRequired state
   367→        onSuccess?.();
   368→      } else {
   369→        setGeneralError(result.error || 'Sign in failed');
   370→        toast.error(result.error || 'Sign in failed');
   371→      }
   372→    },
   373→    [email, password, validate, signIn, onSuccess]
   374→  );
   375→
   376→  return (
   377→    <AuthLayout title="Welcome Back" description="Sign in to your AcquisitionOS account">
   378→      <Card className="glass-card">
   379→        <CardContent className="pt-6 space-y-4">
   380→          <AnimatePresence mode="wait">
   381→            {generalError && (
   382→              <motion.div
   383→                initial={{ opacity: 0, y: -6, height: 0 }}
   384→                animate={{ opacity: 1, y: 0, height: 'auto' }}
   385→                exit={{ opacity: 0, y: -6, height: 0 }}
   386→                className="rounded-lg border border-destructive/30 bg-destructive/5 p-3 text-sm text-destructive overflow-hidden"
   387→                role="alert"
   388→              >
   389→                <div className="flex items-center gap-2">
   390→                  <AlertTriangle className="h-4 w-4 shrink-0" />
   391→                  {generalError}
   392→                </div>
   393→              </motion.div>
   394→            )}
   395→          </AnimatePresence>
   396→
   397→          <form onSubmit={handleSubmit} className="space-y-4" noValidate>
   398→            {/* Email */}
   399→            <div className="space-y-2">
   400→              <Label htmlFor="signin-email" className="text-xs font-medium">
   401→                Email
   402→              </Label>
   403→              <div className={cn('relative transition-all duration-200 rounded-md', errors.email ? 'ring-1 ring-destructive/50' : email && isValidEmail(email) ? 'ring-1 ring-emerald-500/30' : '')}>
   404→                <Mail className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   405→                <Input
   406→                  id="signin-email"
   407→                  type="email"
   408→                  placeholder="you@example.com"
   409→                  className={cn('pl-9 h-10 border-0 bg-muted/50 focus-visible:bg-background transition-colors', errors.email && 'bg-destructive/5')}
   410→                  value={email}
   411→                  onChange={(e) => {
   412→                    setEmail(e.target.value);
   413→                    if (errors.email) setErrors((prev) => ({ ...prev, email: '' }));
   414→                  }}
   415→                  aria-invalid={!!errors.email}
   416→                  aria-describedby={errors.email ? 'signin-email-error' : undefined}
   417→                  autoComplete="email"

   914→export function VerifyEmailPage({
   915→  email,
   916→  onVerified,
   917→  onBackToSignIn,
   918→}: {
   919→  email: string;
   920→  onVerified?: () => void;
   921→  onBackToSignIn?: () => void;
   922→}) {
   923→  const { verifyEmail, signIn } = useAuth();
   924→  const [otp, setOtp] = useState('');
   925→  const [error, setError] = useState('');
   926→  const [loading, setLoading] = useState(false);
   927→  const [resendCooldown, setResendCooldown] = useState(0);
   928→  const cooldownRef = useRef<ReturnType<typeof setInterval>>();
   929→
   930→  // Cooldown timer for resend
   931→  useEffect(() => {
   932→    if (resendCooldown <= 0) return;
   933→    cooldownRef.current = setInterval(() => {
   934→      setResendCooldown((prev) => {
   935→        if (prev <= 1) {
   936→          clearInterval(cooldownRef.current);
   937→          return 0;
   938→        }
   939→        return prev - 1;
   940→      });
   941→    }, 1000);
   942→    return () => clearInterval(cooldownRef.current);
   943→  }, [resendCooldown]);
   944→
   945→  const handleVerify = useCallback(async () => {
   946→    if (otp.length !== 6) {
   947→      setError('Please enter all 6 digits');
   948→      return;
   949→    }
   950→    setError('');
   951→    setLoading(true);
   952→    const result = await verifyEmail(email, otp);
   953→    setLoading(false);
   954→    if (result.success) {
   955→      toast.success('Email verified!');
   956→      onVerified?.();
   957→    } else {
   958→      setError(result.error || 'Verification failed');
   959→      toast.error(result.error || 'Verification failed');
   960→    }
   961→  }, [otp, email, verifyEmail, onVerified]);
   962→
   963→  const handleResend = useCallback(async () => {
   964→    setResendCooldown(60);
   965→    // Call forgot-password-like endpoint to resend
   966→    try {
   967→      const res = await fetch('/api/auth/resend-verification', {
   968→        method: 'POST',
   969→        headers: { 'Content-Type': 'application/json' },
   970→        body: JSON.stringify({ email }),
   971→      });
   972→      if (res.ok) {
   973→        const data = await res.json();
   974→        // Simulation paths permanently disabled — OTP must come via email
   975→        toast.success('Verification code resent!');
   976→      } else {
   977→        toast.error('Failed to resend code');
   978→        setResendCooldown(0);
   979→      }
   980→    } catch {
   981→      toast.error('Network error');
   982→      setResendCooldown(0);
   983→    }
   984→  }, [email]);
   985→
   986→  return (
   987→    <AuthLayout
   988→      title="Verify Your Email"
   989→      description={`Enter the 6-digit code sent to ${email}`}
   990→      showBack
   991→      onBack={onBackToSignIn}
   992→    >
   993→      <Card className="glass-card">
   994→        <CardContent className="pt-6 space-y-6">
   995→          {error && (
   996→            <div className="rounded-lg border border-destructive/30 bg-destructive/5 p-3 text-sm text-destructive slide-in-up" role="alert">
   997→              {error}
   998→            </div>
   999→          )}
  1000→
  1001→          {/* OTP Input */}
  1002→          <div className="flex flex-col items-center gap-2">
  1003→            <Label className="text-xs font-medium text-muted-foreground">Verification Code</Label>
  1004→            <InputOTP maxLength={6} value={otp} onChange={setOtp}>
  1005→              <InputOTPGroup>
  1006→                <InputOTPSlot index={0} className="h-12 w-12 text-lg" />
  1007→                <InputOTPSlot index={1} className="h-12 w-12 text-lg" />
