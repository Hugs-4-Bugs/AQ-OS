   219→   219→// ── OTP Login page ────────────────────────────────────────────
   220→   220→function OtpLoginPage({ onBackToSignIn }: { onBackToSignIn: () => void }) {
   221→   221→  const [email, setEmail] = useState('');
   222→   222→  const [otp, setOtp] = useState('');
   223→   223→  const [step, setStep] = useState<'request' | 'verify'>('request');
   224→   224→  const [loading, setLoading] = useState(false);
   225→   225→  const [error, setError] = useState('');
   226→   226→  const [resendCooldown, setResendCooldown] = useState(0);
   227→   227→  const cooldownRef = useRef<ReturnType<typeof setInterval>>();
   228→   228→
   229→   229→  // Cooldown timer for resend
   230→   230→  useEffect(() => {
   231→   231→    if (resendCooldown <= 0) return;
   232→   232→    cooldownRef.current = setInterval(() => {
   233→   233→      setResendCooldown((prev) => {
   234→   234→        if (prev <= 1) {
   235→   235→          clearInterval(cooldownRef.current);
   236→   236→          return 0;
   237→   237→        }
   238→   238→        return prev - 1;
   239→   239→      });
   240→   240→    }, 1000);
   241→   241→    return () => clearInterval(cooldownRef.current);
   242→   242→  }, [resendCooldown]);
   243→   243→
   244→   244→  const handleRequestOtp = useCallback(
   245→   245→    async (e: React.FormEvent) => {
   246→   246→      e.preventDefault();
   247→   247→      if (!email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   248→   248→        setError('Enter a valid email address');
   249→   249→        return;
   250→   250→      }
   251→   251→      setError('');
   252→   252→      setLoading(true);
   253→   253→      try {
   254→   254→        const res = await fetch('/api/auth/otp/request', {
   255→   255→          method: 'POST',
   256→   256→          headers: { 'Content-Type': 'application/json' },
   257→   257→          body: JSON.stringify({ email }),
   258→   258→        });
   259→   259→        const data = await res.json();
   260→   260→        if (res.ok) {
   261→   261→          setStep('verify');
   262→   262→          setResendCooldown(60);
   263→   263→          // Simulation paths permanently disabled — OTP must come via email
   264→   264→          toast.success('If an account exists for this email, you will receive a sign-in code.');
   265→   265→        } else {
   266→   266→          setError(data.error || 'Failed to send OTP');
   267→   267→          toast.error(data.error || 'Failed to send OTP');
   268→   268→        }
   269→   269→      } catch {
   270→   270→        setError('Network error. Please try again.');
   271→   271→      } finally {
   272→   272→        setLoading(false);
   273→   273→      }
   274→   274→    },
   275→   275→    [email]
   276→   276→  );
   277→   277→
   278→   278→  const handleVerifyOtp = useCallback(async () => {
   279→   279→    if (otp.length !== 6) {
   280→   280→      setError('Enter the 6-digit code');
   281→   281→      return;
   282→   282→    }
   283→   283→    setError('');
   284→   284→    setLoading(true);
   285→   285→    try {
   286→   286→      const res = await fetch('/api/auth/otp/verify', {
   287→   287→        method: 'POST',
   288→   288→        headers: { 'Content-Type': 'application/json' },
   289→   289→        credentials: 'include',
   290→   290→        body: JSON.stringify({ email, otp }),
   291→   291→      });
   292→   292→      const data = await res.json();
   293→   293→      if (res.ok) {
   294→   294→        toast.success('Welcome back!');
   295→   295→        // Auth cookies are set by the API, refresh auth state
   296→   296→        window.location.reload();
   297→   297→      } else {
   298→   298→        setError(data.error || 'Invalid OTP');
   299→   299→        toast.error(data.error || 'Verification failed');
   300→   300→      }
   301→   301→    } catch {
   302→   302→      setError('Network error. Please try again.');
   303→   303→    } finally {
   304→   304→      setLoading(false);
   305→   305→    }
   306→   306→  }, [email, otp]);
   307→   307→
   308→   308→  return (
   309→   309→    <AuthLayout
   310→   310→      title="Sign in with OTP"
   311→   311→      description={step === 'request' ? 'Enter your email to receive a login code' : `Enter the 6-digit code sent to ${email}`}
   312→   312→      showBack
   313→   313→      onBack={onBackToSignIn}
   314→   314→    >
   315→   315→      <Card className="glass-card">
   316→   316→        <CardContent className="pt-6 space-y-4">
   317→   317→          <AnimatePresence mode="wait">
   318→   318→            {error && (
   319→   319→              <motion.div
   320→   320→                initial={{ opacity: 0, y: -6, height: 0 }}
   321→   321→                animate={{ opacity: 1, y: 0, height: 'auto' }}
   322→   322→                exit={{ opacity: 0, y: -6, height: 0 }}
   323→   323→                className="rounded-lg border border-destructive/30 bg-destructive/5 p-3 text-sm text-destructive overflow-hidden"
   324→   324→                role="alert"
   325→   325→              >
   326→   326→                <div className="flex items-center gap-2">
   327→   327→                  <AlertTriangle className="h-4 w-4 shrink-0" />
   328→   328→                  {error}
   329→   329→                </div>
   330→   330→              </motion.div>
   331→   331→            )}
   332→   332→          </AnimatePresence>
   333→   333→
   334→   334→          <AnimatePresence mode="wait">
   335→   335→            {step === 'request' ? (
   336→   336→              <motion.form
   337→   337→                key="request"
   338→   338→                onSubmit={handleRequestOtp}
   339→   339→                className="space-y-4"
   340→   340→                noValidate
   341→   341→                initial={{ opacity: 0, x: -20 }}
   342→   342→                animate={{ opacity: 1, x: 0 }}
   343→   343→                exit={{ opacity: 0, x: 20 }}
   344→   344→                transition={{ duration: 0.2 }}
   345→   345→              >
   346→   346→                <div className="space-y-2">
   347→   347→                  <Label htmlFor="otp-email" className="text-xs font-medium">
   348→   348→                    Email
   349→   349→                  </Label>
   350→   350→                  <div className="relative">
   351→   351→                    <Mail className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   352→   352→                    <Input
   353→   353→                      id="otp-email"
   354→   354→                      type="email"
   355→   355→                      placeholder="you@example.com"
   356→   356→                      className="pl-9 h-10 border-0 bg-muted/50 focus-visible:bg-background transition-colors"
   357→   357→                      value={email}
   358→   358→                      onChange={(e) => { setEmail(e.target.value); if (error) setError(''); }}
   359→   359→                      aria-invalid={!!error}
   360→   360→                      autoComplete="email"
   361→   361→                    />
   362→   362→                  </div>
   363→   363→                </div>
   364→   364→                <Button
   365→   365→                  type="submit"
   366→   366→                  disabled={loading}
   367→   367→                  className="w-full h-10 btn-ripple"
   368→   368→                >
   369→   369→                  {loading ? (
   370→   370→                    <>
   371→   371→                      <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   372→   372→                      Sending...
   373→   373→                    </>
   374→   374→                  ) : (
   375→   375→                    <>
   376→   376→                      <Key className="mr-2 h-4 w-4" />
   377→   377→                      Send Login Code
   378→   378→                    </>
   379→   379→                  )}
   380→   380→                </Button>
   381→   381→              </motion.form>
   382→   382→            ) : (
   383→   383→              <motion.div
   384→   384→                key="verify"
   385→   385→                className="space-y-4"
   386→   386→                initial={{ opacity: 0, x: 20 }}
   387→   387→                animate={{ opacity: 1, x: 0 }}
   388→   388→                exit={{ opacity: 0, x: -20 }}
   389→   389→                transition={{ duration: 0.2 }}
   390→   390→              >
   391→   391→                <div className="flex flex-col items-center gap-2">
   392→   392→                  <Label className="text-xs font-medium text-muted-foreground">Verification Code</Label>
   393→   393→                  <InputOTP maxLength={6} value={otp} onChange={setOtp}>
   394→   394→                    <InputOTPGroup>
   395→   395→                      <InputOTPSlot index={0} className="h-12 w-12 text-lg" />
   396→   396→                      <InputOTPSlot index={1} className="h-12 w-12 text-lg" />
   397→   397→                      <InputOTPSlot index={2} className="h-12 w-12 text-lg" />
   398→   398→                    </InputOTPGroup>
   399→   399→                    <InputOTPSeparator />
   400→   400→                    <InputOTPGroup>
   401→   401→                      <InputOTPSlot index={3} className="h-12 w-12 text-lg" />
   402→   402→                      <InputOTPSlot index={4} className="h-12 w-12 text-lg" />

   404→   404→                    </InputOTPGroup>
   405→   405→                  </InputOTP>
   406→   406→                </div>
   407→   407→                <Button
   408→   408→                  type="button"
   409→   409→                  disabled={loading || otp.length !== 6}
   410→   410→                  onClick={handleVerifyOtp}
   411→   411→                  className="w-full h-10 btn-ripple"
   412→   412→                >
   413→   413→                  {loading ? (
   414→   414→                    <>
   415→   415→                      <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   416→   416→                      Verifying...
   417→   417→                    </>

   419→   419→                    'Verify & Sign In'

   421→   421→                </Button>
   422→   422→                {resendCooldown > 0 ? (
   423→   423→                  <p className="text-xs text-center text-muted-foreground">

   428→   428→                    type="button"

   438→   438→        </CardContent>
