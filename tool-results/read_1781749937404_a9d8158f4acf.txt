    48→/** Password strength calculator */
    49→function getPasswordStrength(password: string): {
    50→  score: number; // 0–4
    51→  label: string;
    52→  color: string;
    53→} {
    54→  if (!password) return { score: 0, label: '', color: '' };
    55→
    56→  let score = 0;
    57→  if (password.length >= 8) score++;
    58→  if (password.length >= 12) score++;
    59→  if (/[A-Z]/.test(password) && /[a-z]/.test(password)) score++;
    60→  if (/\d/.test(password)) score++;
    61→  if (/[^A-Za-z0-9]/.test(password)) score++;
    62→
    63→  // Normalize to 0–4
    64→  score = Math.min(4, score);
    65→
    66→  const labels = ['', 'Weak', 'Fair', 'Good', 'Strong'];
    67→  const colors = [
    68→    '',
    69→    'oklch(0.6 0.25 15)',   // red
    70→    'oklch(0.75 0.18 55)',  // amber
    71→    'oklch(0.85 0.18 90)',  // yellow-green
    72→    'oklch(0.72 0.19 155)', // green
    73→  ];
    74→
    75→  return { score, label: labels[score], color: colors[score] };
    76→}

    78→/** Email validation */
    79→function isValidEmail(email: string): boolean {
    80→  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
    81→}

   300→function PasswordStrengthIndicator({ password }: { password: string }) {
   301→  const strength = getPasswordStrength(password);
   302→  if (!password) return null;
   303→
   304→  const percent = (strength.score / 4) * 100;
   305→
   306→  return (
   307→    <div className="space-y-1.5">
   308→      <Progress
   309→        value={percent}
   310→        className="h-1.5"
   311→        style={
   312→          {
   313→            '--progress-color': strength.color,
   314→          } as React.CSSProperties
   315→        }
   316→      />
   317→      <p className="text-xs" style={{ color: strength.color }}>
   318→        {strength.label}
   319→      </p>
   320→    </div>
   321→  );
   322→}

   360→export function SignInPage({
   361→  onSignUpClick,
   362→  onForgotPasswordClick,
   363→  onSuccess,
   364→  onMagicLinkClick,
   365→  onOtpLoginClick,
   366→}: {
   367→  onSignUpClick?: () => void;
   368→  onForgotPasswordClick?: () => void;
   369→  onSuccess?: () => void;
   370→  onMagicLinkClick?: () => void;
   371→  onOtpLoginClick?: () => void;
   372→}) {
   373→  const { signIn, isLoading } = useAuth();
   374→  const [email, setEmail] = useState('');
   375→  const [password, setPassword] = useState('');
   376→  const [showPassword, setShowPassword] = useState(false);
   377→  const [errors, setErrors] = useState<Record<string, string>>({});
   378→  const [generalError, setGeneralError] = useState('');
   379→  const [authConfig, setAuthConfig] = useState<{ googleAvailable: boolean; loaded: boolean }>({ googleAvailable: false, loaded: false });
   380→
   381→  // Fetch auth config on mount to check Google OAuth availability
   382→  useEffect(() => {
   383→    fetch('/api/auth/config')
   384→      .then(res => res.ok ? res.json() : { googleAvailable: false })
   385→      .then(data => setAuthConfig({ googleAvailable: !!data.googleAvailable, loaded: true }))
   386→      .catch(() => setAuthConfig({ googleAvailable: false, loaded: true }));
   387→  }, []);
   388→
   389→  // Server-driven Google OAuth: client_id never touches frontend
   390→  const startGoogleOAuth = useCallback(async () => {
   391→    try {
   392→      const origin = typeof window !== 'undefined' ? window.location.origin : '';
   393→      const res = await fetch(`/api/auth/google/state?origin=${encodeURIComponent(origin)}`);
   394→      if (!res.ok) throw new Error('Failed to get OAuth state');
   395→      const data = await res.json();
   396→      if (data.authUrl) {
   397→        window.location.href = data.authUrl;
   398→      }
   399→    } catch (err) {
   400→      console.error('Google OAuth error:', err);
   401→      toast.error('Google sign-in is not available in this environment.');
   402→    }
   403→  }, []);
   404→
   405→  const validate = useCallback(() => {
   406→    const errs: Record<string, string> = {};
   407→    if (!email) errs.email = 'Email is required';
   408→    else if (!isValidEmail(email)) errs.email = 'Enter a valid email address';
   409→    if (!password) errs.password = 'Password is required';
   410→    else if (password.length < 6) errs.password = 'Password must be at least 6 characters';
   411→    return errs;
   412→  }, [email, password]);
   413→
   414→  const handleSubmit = useCallback(
   415→    async (e: React.FormEvent) => {
   416→      e.preventDefault();
   417→      setGeneralError('');
   418→      const validationErrors = validate();
   419→      setErrors(validationErrors);
   420→      if (Object.keys(validationErrors).length > 0) return;
   421→
   422→      const result = await signIn(email, password);
   423→      if (result.success) {
   424→        toast.success('Welcome back!');
   425→        onSuccess?.();
   426→      } else if (result.mfaRequired) {
   427→        toast.info('MFA verification required');
   428→        // Caller should handle MFA redirect via mfaRequired state
   429→        onSuccess?.();
   430→      } else {
   431→        setGeneralError(result.error || 'Sign in failed');
   432→        toast.error(result.error || 'Sign in failed');
   433→      }
   434→    },
   435→    [email, password, validate, signIn, onSuccess]
   436→  );
   437→
   438→  return (
   439→    <AuthLayout title="Welcome Back" description="Sign in to your AcquisitionOS account">
   440→      <Card className="glass-card">
   441→        <CardContent className="pt-6 space-y-4">
   442→          <AnimatePresence mode="wait">
   443→            {generalError && (
   444→              <motion.div
   445→                initial={{ opacity: 0, y: -6, height: 0 }}
   446→                animate={{ opacity: 1, y: 0, height: 'auto' }}
   447→                exit={{ opacity: 0, y: -6, height: 0 }}
   448→                className="rounded-lg border border-destructive/30 bg-destructive/5 p-3 text-sm text-destructive overflow-hidden"
   449→                role="alert"
   450→              >
   451→                <div className="flex items-center gap-2">
   452→                  <AlertTriangle className="h-4 w-4 shrink-0" />
   453→                  {generalError}
   454→                </div>
   455→              </motion.div>
   456→            )}
   457→          </AnimatePresence>
   458→
   459→          <form onSubmit={handleSubmit} className="space-y-4" noValidate>
   460→            {/* Email */}
   461→            <div className="space-y-2">
   462→              <Label htmlFor="signin-email" className="text-xs font-medium">
   463→                Email
   464→              </Label>
   465→              <div className={cn('relative transition-all duration-200 rounded-md', errors.email ? 'ring-1 ring-destructive/50' : email && isValidEmail(email) ? 'ring-1 ring-emerald-500/30' : '')}>
   466→                <Mail className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   467→                <Input
   468→                  id="signin-email"
   469→                  type="email"
   470→                  placeholder="you@example.com"
   471→                  className={cn('pl-9 h-10 border-0 bg-muted/50 focus-visible:bg-background transition-colors', errors.email && 'bg-destructive/5')}
   472→                  value={email}
   473→                  onChange={(e) => {
   474→                    setEmail(e.target.value);
   475→                    if (errors.email) setErrors((prev) => ({ ...prev, email: '' }));
   476→                  }}
   477→                  aria-invalid={!!errors.email}
   478→                  aria-describedby={errors.email ? 'signin-email-error' : undefined}
   479→                  autoComplete="email"
   480→                />
   481→                {email && isValidEmail(email) && !errors.email && (
   482→                  <Check className="absolute right-3 top-1/2 -translate-y-1/2 h-4 w-4 text-emerald-500" />
