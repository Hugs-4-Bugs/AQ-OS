    15→async function emitWorkflowEvent(
    16→  eventType: string,
    17→  userId: string,
    18→  payload: Record<string, unknown>
    19→): Promise<void> {
    20→  try {
    21→    await publishEvent({
    22→      channel: 'workflow_events',
    23→      eventType,
    24→      payload,
    25→      userId,
    26→    });
    27→  } catch {
    28→    // Non-blocking: don't fail execution if event bus fails
    29→  }
    30→}

    57→// ===== MAIN EXECUTION ENTRY POINT =====
    58→
    59→/**
    60→ * Execute a workflow. Creates a WorkflowExecution record and processes steps.
    61→ * Returns the execution ID immediately (202 Accepted pattern).
    62→ */
    63→export async function executeWorkflow(
    64→  workflowId: string,
    65→  userId: string,
    66→  triggerData?: Record<string, unknown>,
    67→  idempotencyKey?: string
    68→): Promise<ExecuteWorkflowResult> {
    69→  try {
    70→    // Load workflow definition — allow both 'draft' (for testing) and 'active' workflows.
    71→    // Drafts can be executed to validate the workflow before publishing.
    72→    const workflow = await db.workflowDefinition.findFirst({
    73→      where: { id: workflowId, userId, status: { in: ['draft', 'active'] } },
    74→      include: { workflowSteps: { orderBy: { order: 'asc' } } },
    75→    });
    76→
    77→    if (!workflow) {
    78→      return { executionId: '', status: 'failed', error: 'Workflow not found or not active (drafts are executable for testing)' };
    79→    }
    80→
    81→    // Check execution limit
    82→    const limitCheck = await checkExecutionLimit(userId);
    83→    if (!limitCheck.allowed) {
    84→      return {
    85→        executionId: '',
    86→        status: 'failed',
    87→        error: `Monthly execution limit reached (${limitCheck.current}/${limitCheck.limit})`,
    88→      };
    89→    }
    90→
    91→    // Idempotency check
    92→    if (idempotencyKey) {
    93→      const existing = await db.workflowExecution.findFirst({
    94→        where: { idempotencyKey, workflowId },
    95→      });
    96→      if (existing) {
    97→        return { executionId: existing.id, status: existing.status };
    98→      }
    99→    }
   100→
   101→    // Parse steps from nodes/edges if no explicit steps
   102→    let steps: StepDefinition[] = [];
   103→    if (workflow.workflowSteps.length > 0) {
   104→      steps = workflow.workflowSteps.map((s) => ({
   105→        id: s.id,
   106→        type: s.type,
   107→        name: s.name,
   108→        config: JSON.parse(s.config || '{}'),
   109→        order: s.order,
   110→        nextStepId: s.nextStepId || undefined,
   111→      }));
   112→    } else {
   113→      // Parse nodes from JSON
   114→      try {
   115→        const nodes = JSON.parse(workflow.nodes || '[]');
   116→        steps = nodes
   117→          .filter((n: Record<string, unknown>) => n.type !== 'trigger')
   118→          .map((n: Record<string, unknown>, i: number) => ({
   119→            id: String(n.id || `step-${i}`),
   120→            type: String(n.type || 'action'),
   121→            name: String(n.title || n.name || `Step ${i + 1}`),
   122→            config: (n.config as Record<string, unknown>) || {},
   123→            order: i,
   124→          }));
   125→      } catch {
   126→        steps = [];
   127→      }
   128→    }
   129→
   130→    // Create execution record
   131→    const maxRetries = Number(process.env.WORKFLOW_MAX_RETRIES) || 3;
   132→    const timeoutMs = Number(process.env.WORKFLOW_TIMEOUT) || 300000;
   133→
   134→    const execution = await db.workflowExecution.create({
   135→      data: {
   136→        workflowId,
   137→        // Persist the execution owner. The trusted execution context (userId +
   138→        // leadId) must survive on the execution record so that retries,
   139→        // resumes, and audit trails all resolve leads against the SAME user
   140→        // who started the run — never against a null/system owner.
   141→        userId,
   142→        status: 'running',
   143→        triggerData: triggerData ? JSON.stringify(triggerData) : null,
   144→        triggerEvent: triggerData ? JSON.stringify(triggerData) : null,
   145→        currentStep: 0,
   146→        totalSteps: steps.length,
   147→        idempotencyKey: idempotencyKey || null,
   148→        maxRetries,
   149→        timeoutMs,
   150→      },
   151→    });
   152→
   153→    // Execute steps (async but we process them sequentially)
   154→    // We do NOT await the full execution to return 202 quickly
   155→    // But in this single-process model, we run inline
   156→    try {
   157→      await runSteps(execution.id, steps, userId, workflowId, triggerData);
   158→    } catch (error) {
   159→      console.error('[WorkflowEngine] Step execution error:', error);
   160→    }
   161→
   162→    // Log audit event
   163→    await logWorkflowEvent(userId, 'workflow_executed', {
   164→      workflowId,
   165→      executionId: execution.id,
   166→      triggerType: workflow.triggerType,
   167→    });
   168→
   169→    // Emit realtime event
   170→    await emitWorkflowEvent('workflow_started', userId, {
   171→      workflowId,
   172→      executionId: execution.id,
   173→      workflowName: workflow.name,
   174→      triggerType: workflow.triggerType,
   175→    });
   176→
   177→    // Fetch final state
   178→    const finalExecution = await db.workflowExecution.findUnique({
   179→      where: { id: execution.id },
   180→      select: { id: true, status: true, error: true },
   181→    });
   182→
   183→    return {
   184→      executionId: execution.id,
   185→      status: finalExecution?.status || 'running',
   186→      error: finalExecution?.error || undefined,
   187→    };
   188→  } catch (error) {
   189→    console.error('[WorkflowEngine] Failed to execute workflow:', error);
   190→    return {
   191→      executionId: '',
   192→      status: 'failed',
   193→      error: error instanceof Error ? error.message : 'Execution failed',
   194→    };
   195→  }
   196→}

   198→// ===== STEP RUNNER =====
   199→
   200→async function runSteps(
   201→  executionId: string,
   202→  steps: StepDefinition[],
   203→  userId: string,
   204→  workflowId: string,
   205→  triggerData?: Record<string, unknown>
   206→): Promise<void> {
   207→  const previousOutputs: Record<string, unknown> = {};
   208→  let shouldContinue = true;
   209→
   210→  for (let i = 0; i < steps.length && shouldContinue; i++) {
   211→    const step = steps[i];
   212→
   213→    // Update current step
   214→    await db.workflowExecution.update({
   215→      where: { id: executionId },
   216→      data: { currentStep: i },
   217→    });
   218→
   219→    // Process the step
   220→    const result = await processStep(executionId, step, {
   221→      userId,
   222→      workflowId,
   223→      executionId,
   224→      leadId: triggerData?.leadId as string | undefined,
   225→      triggerData,
   226→      previousOutputs,
   227→    });
   228→
   229→    // Store output
   230→    if (result.output) {
   231→      previousOutputs[step.id] = result.output;
   232→    }
   233→
   234→    // Handle delay steps
   235→    if (step.type === 'delay' || step.type === 'wait_delay') {
   236→      const delayMs = result.output?.delayMs as number | undefined;
   237→      if (delayMs && delayMs > 0 && delayMs <= 86400000) {
   238→        // For short delays (< 24h), sleep inline
   239→        // For longer delays, this should be handled by a queue
   240→        await new Promise(resolve => setTimeout(resolve, Math.min(delayMs, 60000)));
   241→      }
   242→    }
   243→
   244→    // Handle conditional branch — may skip subsequent steps
   245→    if (step.type === 'condition' || step.type === 'conditional_branch') {
   246→      const branch = result.output?.branch as string | undefined;
   247→      if (branch === 'false' || branch === 'skip') {
   248→        // Skip the next step if condition is false
   249→        // (Simple linear branch model)
   250→        i++; // Skip next step
   251→      }
   252→    }
   253→
   254→    if (!result.success) {
   255→      // Mark execution as failed
   256→      await db.workflowExecution.update({
   257→        where: { id: executionId },
   258→        data: {
   259→          status: 'failed',
   260→          error: result.error || 'Step execution failed',
   261→          completedAt: new Date(),
   262→        },
   263→      });
   264→
   265→      // User-facing notification (safe message — raw errors stay in the
   266→      // execution record, never in the notification text).
   267→      await createNotificationOnce({
   268→        userId,
   269→        type: 'workflow_failed',
   270→        title: 'Workflow execution failed',
   271→        message: `A workflow run stopped at step "${step.name || 'unknown'}" and could not continue. You can retry it from the Workflows page.`,
   272→        actionUrl: '/business-ai/workflows',
   273→        metadata: { workflowId, executionId, failedStep: step.name },
   274→        dedupeKey: `wfexec:${executionId}:failed`,
   275→      }).catch(() => {
   276→        // Never fail the execution path because of a notification problem
   277→      });
   278→
   279→      await logWorkflowEvent(userId, 'workflow_failed', {
   280→        workflowId,
   281→        executionId,
   282→        failedStep: step.name,
   283→        error: result.error,
   284→      });
   285→
   286→      // Emit realtime event
   287→      await emitWorkflowEvent('workflow_failed', userId, {
   288→        workflowId,
   289→        executionId,
   290→        failedStep: step.name,
   291→        error: result.error,
   292→      });
   293→
   294→      // Check if we should retry
   295→      const execution = await db.workflowExecution.findUnique({
   296→        where: { id: executionId },
   297→        select: { retryCount: true, maxRetries: true },
   298→      });
   299→
   300→      if (execution && execution.retryCount < execution.maxRetries) {
   301→        await db.workflowExecution.update({
   302→          where: { id: executionId },
   303→          data: { retryCount: { increment: 1 } },
   304→        });
   305→      } else {
   306→        // Max retries exceeded — dead letter
   307→        await sendToDeadLetter(executionId, `Max retries exceeded. Last error: ${result.error || 'Unknown'}`);
   308→      }
   309→
   310→      return;
   311→    }
   312→
   313→    shouldContinue = result.shouldContinue;
   314→  }
   315→
   316→  // Mark execution as completed
   317→  if (shouldContinue) {
   318→    await db.workflowExecution.update({
   319→      where: { id: executionId },
   320→      data: {
   321→        status: 'completed',
   322→        completedAt: new Date(),
   323→        currentStep: steps.length,
   324→      },
   325→    });
   326→
   327→    // User-facing notification (deduped per execution — replay/retry of the
   328→    // same execution id will not create a second notification).
   329→    await createNotificationOnce({
   330→      userId,
   331→      type: 'workflow_completed',
   332→      title: 'Workflow completed',
   333→      message: `A workflow run finished successfully — all ${steps.length} step${steps.length === 1 ? '' : 's'} completed.`,
   334→      actionUrl: '/business-ai/workflows',
   335→      metadata: { workflowId, executionId, totalSteps: steps.length },
   336→      dedupeKey: `wfexec:${executionId}:completed`,
   337→    }).catch(() => {
   338→      // Never fail the execution path because of a notification problem
   339→    });
   340→
   341→    // Emit realtime completion event
   342→    await emitWorkflowEvent('workflow_completed', userId, {
   343→      workflowId,
   344→      executionId,
   345→      totalSteps: steps.length,
   346→    });
   347→  }
   348→}

   350→// ===== PROCESS SINGLE STEP =====
   351→
   352→export async function processStep(
   353→  executionId: string,
   354→  step: StepDefinition,
   355→  context: ActionContext
   356→): Promise<ProcessStepResult> {
   357→  const startTime = Date.now();
   358→
   359→  try {
   360→    let result: ActionResult;
   361→
   362→    // Deduct credits before execution
   363→    const creditResult = await deductExecutionCredits(context.userId, step.type);
   364→
   365→    if (!creditResult.success && creditResult.creditsUsed > 0) {
   366→      // Only fail if the action actually costs credits and we can't afford them
   367→      const durationMs = Date.now() - startTime;
   368→
   369→      await db.workflowLog.create({
   370→        data: {
   371→          executionId,
   372→          stepName: step.name,
   373→          stepType: step.type,
   374→          status: 'failed',
   375→          input: JSON.stringify(step.config),
   376→          error: `Insufficient credits: ${creditResult.error}`,
   377→          durationMs,
   378→          userId: context.userId,
   379→        },
   380→      });
   381→
   382→      return {
   383→        success: false,
   384→        error: `Insufficient credits: ${creditResult.error}`,
   385→        shouldContinue: false,
   386→        durationMs,
   387→      };
   388→    }
   389→
   390→    // Execute the action.
   391→    // WorkflowStep.type stores the NODE type ('action' | 'delay' | 'condition' | 'ai_action'),
   392→    // while executeAction dispatches on the ACTION type ('send_email' | 'wait_delay' | ...).
   393→    // The action type always rides in config.actionType; fall back to step.type for
   394→    // legacy steps that stored the action type directly.
   395→    const actionType = String(step.config?.actionType || step.type);
   396→    result = await executeAction(actionType, step.config, context);
   397→
   398→    const durationMs = Date.now() - startTime;
   399→
   400→    // Create workflow log entry
   401→    await db.workflowLog.create({
   402→      data: {
   403→        executionId,
   404→        stepName: step.name,
   405→        stepType: step.type,
   406→        status: result.success ? 'success' : 'failed',
   407→        input: JSON.stringify(step.config),
   408→        output: result.output ? JSON.stringify(result.output) : null,
   409→        error: result.error || null,
   410→        durationMs,
   411→        userId: context.userId,
   412→      },
   413→    });
   414→
   415→    // Emit step event
   416→    await emitWorkflowEvent(
   417→      result.success ? 'workflow_step_completed' : 'workflow_step_failed',
   418→      context.userId,
   419→      {
   420→        executionId,
   421→        stepName: step.name,
   422→        stepType: step.type,
   423→        durationMs,
   424→        error: result.error,
   425→      }
   426→    );
   427→
   428→    return {
   429→      success: result.success,
   430→      output: result.output,
   431→      error: result.error,
   432→      shouldContinue: result.success,
   433→      durationMs,
   434→    };
   435→  } catch (error) {
   436→    const durationMs = Date.now() - startTime;
   437→
   438→    await db.workflowLog.create({
   439→      data: {
   440→        executionId,
   441→        stepName: step.name,
   442→        stepType: step.type,
   443→        status: 'failed',
   444→        input: JSON.stringify(step.config),
   445→        error: error instanceof Error ? error.message : 'Step execution failed',
   446→        durationMs,
   447→        userId: context.userId,
   448→      },
   449→    });
   450→
   451→    return {
   452→      success: false,
   453→      error: error instanceof Error ? error.message : 'Step execution failed',
   454→      shouldContinue: false,
   455→      durationMs,
   456→    };
   457→  }
   458→}

   460→// ===== PAUSE EXECUTION =====
   461→
   462→export async function pauseExecution(
   463→  executionId: string,
   464→  userId: string
   465→): Promise<{ success: boolean; error?: string }> {
   466→  try {
   467→    const execution = await db.workflowExecution.findFirst({
   468→      where: { id: executionId },
   469→      include: { workflow: { select: { userId: true } } },
   470→    });
   471→
   472→    if (!execution) {
   473→      return { success: false, error: 'Execution not found' };
   474→    }
   475→
   476→    if (execution.workflow.userId !== userId) {
   477→      return { success: false, error: 'Not authorized' };
   478→    }
   479→
   480→    if (execution.status !== 'running') {
   481→      return { success: false, error: `Cannot pause execution in '${execution.status}' status` };
   482→    }
   483→
   484→    await db.workflowExecution.update({
   485→      where: { id: executionId },
   486→      data: { status: 'paused', pausedAt: new Date() },
   487→    });
   488→
   489→    await logWorkflowEvent(userId, 'workflow_paused', { executionId });
   490→
   491→    // Emit realtime event
   492→    await emitWorkflowEvent('workflow_paused', userId, { executionId });
   493→
   494→    return { success: true };
   495→  } catch (error) {
   496→    return {
   497→      success: false,
   498→      error: error instanceof Error ? error.message : 'Failed to pause execution',
   499→    };
   500→  }
   501→}

   503→// ===== RESUME EXECUTION =====
   504→
   505→export async function resumeExecution(
   506→  executionId: string,
   507→  userId: string
   508→): Promise<{ success: boolean; error?: string }> {
   509→  try {
   510→    const execution = await db.workflowExecution.findFirst({
   511→      where: { id: executionId },
   512→      include: { workflow: { select: { userId: true } } },
   513→    });
   514→
   515→    if (!execution) {
   516→      return { success: false, error: 'Execution not found' };
   517→    }
   518→
   519→    if (execution.workflow.userId !== userId) {
   520→      return { success: false, error: 'Not authorized' };
   521→    }
   522→
   523→    if (execution.status !== 'paused') {
   524→      return { success: false, error: `Cannot resume execution in '${execution.status}' status` };
   525→    }
   526→
   527→    await db.workflowExecution.update({
   528→      where: { id: executionId },
   529→      data: { status: 'running', resumedAt: new Date() },
   530→    });
   531→
   532→    // Re-run remaining steps
   533→    const workflow = await db.workflowDefinition.findUnique({
   534→      where: { id: execution.workflowId },
   535→      include: { workflowSteps: { orderBy: { order: 'asc' } } },
   536→    });
   537→
   538→    if (workflow) {
   539→      const steps = workflow.workflowSteps.map((s) => ({
   540→        id: s.id,
   541→        type: s.type,
   542→        name: s.name,
   543→        config: JSON.parse(s.config || '{}'),
   544→        order: s.order,
   545→        nextStepId: s.nextStepId || undefined,
   546→      }));
   547→
   548→      const triggerData = execution.triggerData ? JSON.parse(execution.triggerData) : undefined;
   549→      const remainingSteps = steps.slice(execution.currentStep);
   550→
   551→      await runSteps(executionId, remainingSteps, userId, execution.workflowId, triggerData);
   552→    }
   553→
   554→    await logWorkflowEvent(userId, 'workflow_resumed', { executionId });
   555→
   556→    // Emit realtime event
   557→    await emitWorkflowEvent('workflow_resumed', userId, { executionId });
   558→
   559→    return { success: true };
   560→  } catch (error) {
   561→    return {
   562→      success: false,
   563→      error: error instanceof Error ? error.message : 'Failed to resume execution',
   564→    };
   565→  }
   566→}

   607→    };
   608→  }

   611→// ===== RETRY EXECUTION =====
   612→
   613→export async function retryExecution(
   614→  executionId: string,
   615→  userId: string
   616→): Promise<{ success: boolean; executionId?: string; error?: string }> {
   617→  try {
   618→    const execution = await db.workflowExecution.findFirst({
   619→      where: { id: executionId },
   620→      include: { workflow: { select: { userId: true, id: true } } },
   621→    });
   622→
   623→    if (!execution) {
   624→      return { success: false, error: 'Execution not found' };
   625→    }
   626→
   627→    if (execution.workflow.userId !== userId) {
   628→      return { success: false, error: 'Not authorized' };
   629→    }
   630→
   631→    if (!['failed', 'cancelled'].includes(execution.status)) {
   632→      return { success: false, error: `Cannot retry execution in '${execution.status}' status` };
   633→    }
   634→
   635→    // Re-execute from the failed step
   636→    const workflow = await db.workflowDefinition.findUnique({
   637→      where: { id: execution.workflowId },
   638→      include: { workflowSteps: { orderBy: { order: 'asc' } } },
   639→    });
   640→
   641→    if (!workflow) {
   642→      return { success: false, error: 'Workflow not found' };
   643→    }
   644→
   645→    const steps = workflow.workflowSteps.map((s) => ({
   646→      id: s.id,
   647→      type: s.type,
   648→      name: s.name,
   649→      config: JSON.parse(s.config || '{}'),
   650→      order: s.order,
   651→      nextStepId: s.nextStepId || undefined,
   652→    }));
   653→
   654→    // Reset execution state
   655→    await db.workflowExecution.update({
   656→      where: { id: executionId },
   657→      data: {
   658→        status: 'running',
   659→        error: null,
   660→        completedAt: null,
   661→        retryCount: { increment: 1 },
   662→      },
   663→    });
   664→
   665→    const triggerData = execution.triggerData ? JSON.parse(execution.triggerData) : undefined;
   666→    const remainingSteps = steps.slice(execution.currentStep);
   667→
   668→    await runSteps(executionId, remainingSteps, userId, execution.workflowId, triggerData);
   669→
   670→    await logWorkflowEvent(userId, 'workflow_retried', {
   671→      executionId,
   672→      workflowId: execution.workflowId,
   673→    });
   674→
   675→    // Emit realtime event
   676→    await emitWorkflowEvent('workflow_retried', userId, {
   677→      executionId,
   678→      workflowId: execution.workflowId,
   679→    });
   680→
   681→    return { success: true, executionId };
   682→  } catch (error) {
   683→    return {
   684→      success: false,
   685→      error: error instanceof Error ? error.message : 'Failed to retry execution',
   686→    };
   687→  }
   688→}

   690→// ===== EXECUTION HISTORY =====
   691→
   692→export async function getExecutionHistory(
   693→  userId: string,
   694→  filters?: {
   695→    status?: string;
   696→    workflowId?: string;
   697→    dateFrom?: string;
   698→    dateTo?: string;
   699→    page?: number;
   700→    limit?: number;
   701→  }
   702→): Promise<{ executions: Record<string, unknown>[]; total: number; page: number; totalPages: number }> {
   703→  try {
   704→    const page = filters?.page || 1;
   705→    const limit = Math.min(filters?.limit || 20, 100);
   706→    const offset = (page - 1) * limit;
   707→
   708→    const where: Record<string, unknown> = {
   709→      workflow: { userId },
   710→    };
   711→
   712→    if (filters?.status) where.status = filters.status;
   713→    if (filters?.workflowId) where.workflowId = filters.workflowId;
   714→    if (filters?.dateFrom || filters?.dateTo) {
   715→      const startedAt: Record<string, Date> = {};
   716→      if (filters?.dateFrom) startedAt.gte = new Date(filters.dateFrom);
   717→      if (filters?.dateTo) startedAt.lte = new Date(filters.dateTo);
   718→      where.startedAt = startedAt;
   719→    }
   720→
   721→    const [executions, total] = await Promise.all([
   722→      db.workflowExecution.findMany({
   723→        where,
   724→        orderBy: { startedAt: 'desc' },
   725→        skip: offset,
   726→        take: limit,
   727→        include: {
   728→          workflow: { select: { name: true, triggerType: true } },
   729→        },
   730→      }),
   731→      db.workflowExecution.count({ where }),
   732→    ]);
   733→
   734→    return {
   735→      executions: executions.map((e) => ({
   736→        id: e.id,
   737→        workflowId: e.workflowId,
   738→        workflowName: e.workflow.name,
   739→        triggerType: e.workflow.triggerType,
   740→        status: e.status,
   741→        currentStep: e.currentStep,
   742→        totalSteps: e.totalSteps,
   743→        startedAt: e.startedAt,
   744→        completedAt: e.completedAt,
   745→        error: e.error,
   746→        retryCount: e.retryCount,
   747→        deadLettered: e.deadLettered,
   748→        deadLetterReason: e.deadLetterReason,
   749→      })),
   750→      total,
   751→      page,
   752→      totalPages: Math.ceil(total / limit),
   753→    };
   754→  } catch (error) {
   755→    console.error('[WorkflowEngine] Failed to get execution history:', error);
   756→    return { executions: [], total: 0, page: 1, totalPages: 0 };
   757→  }
   758→}

   818→// ===== EXECUTION METRICS =====
   819→
   820→export async function getExecutionMetrics(
   821→  userId: string
   822→): Promise<Record<string, unknown>> {
   823→  try {
   824→    // Total executions
   825→    const total = await db.workflowExecution.count({
   826→      where: { workflow: { userId } },
   827→    });
   828→
   829→    // Success count
   830→    const completed = await db.workflowExecution.count({
   831→      where: { workflow: { userId }, status: 'completed' },
   832→    });
   833→
   834→    // Failed count
   835→    const failed = await db.workflowExecution.count({
   836→      where: { workflow: { userId }, status: 'failed' },
   837→    });
   838→
   839→    // Running count
   840→    const running = await db.workflowExecution.count({
   841→      where: { workflow: { userId }, status: 'running' },
   842→    });
   843→
   844→    // Dead lettered count
   845→    const deadLettered = await db.workflowExecution.count({
   846→      where: { workflow: { userId }, deadLettered: true },
   847→    });
   848→
   849→    // Success rate
   850→    const successRate = total > 0 ? ((completed / total) * 100).toFixed(1) : '0';
   851→
   852→    // Average duration (for completed executions)
   853→    const completedExecutions = await db.workflowExecution.findMany({
   854→      where: { workflow: { userId }, status: 'completed', completedAt: { not: null } },
   855→      select: { startedAt: true, completedAt: true },
   856→      take: 100,
   857→    });
   858→
   859→    const durations = completedExecutions
   860→      .filter((e) => e.completedAt)
   861→      .map((e) => e.completedAt!.getTime() - e.startedAt.getTime());
   862→
   863→    const avgDurationMs = durations.length > 0
   864→      ? Math.round(durations.reduce((a, b) => a + b, 0) / durations.length)
   865→      : 0;
   866→
   867→    // By trigger type
   868→    const byTriggerType = await db.workflowExecution.groupBy({
   869→      by: ['workflowId'],
   870→      where: { workflow: { userId } },
   871→      _count: { id: true },
   872→    });
   873→
   874→    // By status
   875→    const byStatus = await db.workflowExecution.groupBy({
   876→      by: ['status'],
   877→      where: { workflow: { userId } },
   878→      _count: { id: true },
   879→    });
   880→
   881→    // This month
   882→    const startOfMonth = new Date(new Date().getFullYear(), new Date().getMonth(), 1);
   883→    const thisMonth = await db.workflowExecution.count({
   884→      where: { workflow: { userId }, startedAt: { gte: startOfMonth } },
   885→    });
   886→
   887→    return {
   888→      total,
   889→      completed,
   890→      failed,
   891→      running,
   892→      deadLettered,
   893→      successRate: `${successRate}%`,
   894→      avgDurationMs,
   895→      avgDurationFormatted: formatDuration(avgDurationMs),
   896→      thisMonth,
   897→      byStatus: byStatus.map((s) => ({ status: s.status, count: s._count.id })),
   898→      byTriggerType: byTriggerType.map((t) => ({ workflowId: t.workflowId, count: t._count.id })),
   899→    };
   900→  } catch (error) {
   901→    console.error('[WorkflowEngine] Failed to get execution metrics:', error);
   902→    return {
   903→      total: 0,
   904→      completed: 0,
   905→      failed: 0,
   906→      running: 0,
   907→      deadLettered: 0,
   908→      successRate: '0%',
   909→      avgDurationMs: 0,
   910→      thisMonth: 0,
   911→    };
   912→  }
   913→}
   914→
   915→function formatDuration(ms: number): string {
   916→  if (ms < 1000) return `${ms}ms`;
   917→  if (ms < 60000) return `${(ms / 1000).toFixed(1)}s`;
   918→  if (ms < 3600000) return `${(ms / 60000).toFixed(1)}m`;
   919→  return `${(ms / 3600000).toFixed(1)}h`;
   920→}
