   200→/** Human label for the parsed hard criteria (badges + honest notices). */
   201→function describeCriteria(criteria?: ParsedIntent['criteria']): string {
   202→  if (!criteria) return '';
   203→  const parts: string[] = [];
   204→  if (criteria.exactEmployeeCount != null) parts.push(`exactly ${criteria.exactEmployeeCount} employees`);
   205→  else if (criteria.employeeMin != null && criteria.employeeMax != null) parts.push(`${criteria.employeeMin}–${criteria.employeeMax} employees`);
   206→  else if (criteria.employeeMin != null) parts.push(`≥ ${criteria.employeeMin} employees`);
   207→  else if (criteria.employeeMax != null) parts.push(`≤ ${criteria.employeeMax} employees`);
   208→  if (criteria.website === 'required') parts.push('website required');
   209→  if (criteria.website === 'absent') parts.push('no website');
   210→  if (criteria.excludeTypes?.length) parts.push(`excludes: ${criteria.excludeTypes.join(', ')}`);
   211→  return parts.join(' · ');
   212→}

   227→function formatRelativeTime(dateStr: string): string {
   228→  const date = new Date(dateStr);
   229→  const now = new Date();
   230→  const diffMs = now.getTime() - date.getTime();
   231→  const diffMins = Math.floor(diffMs / 60000);
   232→  const diffHours = Math.floor(diffMins / 60);
   233→  const diffDays = Math.floor(diffHours / 24);
   234→  if (diffMins < 1) return 'Just now';
   235→  if (diffMins < 60) return `${diffMins}m ago`;
   236→  if (diffHours < 24) return `${diffHours}h ago`;
   237→  if (diffDays === 1) return 'Yesterday';
   238→  if (diffDays < 7) return `${diffDays}d ago`;
   239→  return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
   240→}

   321→// ─── Discovery Job Progress Card ─────────────────────────
   322→function DiscoveryJobProgress({
   323→  job,
   324→  onComplete,
   325→}: {
   326→  job: DiscoveryJob;
   327→  onComplete: (job: DiscoveryJob) => void;
   328→}) {
   329→  const [currentJob, setCurrentJob] = useState<DiscoveryJob>(job);
   330→  const intervalRef = useRef<NodeJS.Timeout | null>(null);
   331→
   332→  useEffect(() => {
   333→    if (currentJob.status !== 'running' && currentJob.status !== 'pending') return;
   334→
   335→    intervalRef.current = setInterval(async () => {
   336→      try {
   337→        const res = await fetch(`/api/leads/discover/status/${currentJob.id}`);
   338→        if (res.ok) {
   339→          const data = await res.json();
   340→          const updatedJob = normalizeJobFromApi(data.job);
   341→          setCurrentJob(updatedJob);
   342→
   343→          if (updatedJob.status === 'completed' || updatedJob.status === 'failed') {
   344→            if (intervalRef.current) clearInterval(intervalRef.current);
   345→            onComplete(updatedJob);
   346→
   347→            // Update localStorage
   348→            try {
   349→              const stored = localStorage.getItem('acquisitionos_discovery_jobs');
   350→              const jobs: DiscoveryJob[] = stored ? JSON.parse(stored) : [];
   351→              const idx = jobs.findIndex(j => j.id === updatedJob.id);
   352→              if (idx >= 0) jobs[idx] = updatedJob;
   353→              else jobs.unshift(updatedJob);
   354→              localStorage.setItem('acquisitionos_discovery_jobs', JSON.stringify(jobs.slice(0, 20)));
   355→            } catch {}
   356→          }
   357→        }
   358→      } catch {}
   359→    }, 3000);
   360→
   361→    return () => {
   362→      if (intervalRef.current) clearInterval(intervalRef.current);
   363→    };
   364→  }, [currentJob.id, currentJob.status, onComplete]);
   365→
   366→  const progress = currentJob.status === 'completed' ? 100 :
   367→    currentJob.status === 'running' ? 60 :
   368→    currentJob.status === 'pending' ? 10 : 0;
   369→
   370→  const progressSteps = [
   371→    { label: 'Connecting', done: progress >= 10 },
   372→    { label: 'Searching', done: progress >= 40 },
   373→    { label: 'Analyzing', done: progress >= 60 },
   374→    { label: 'Complete', done: progress >= 100 },
   375→  ];
   376→
   377→  return (
   378→    <Card className="border-primary/20 overflow-hidden">
   379→      <CardContent className="p-4 space-y-3">
   380→        <div className="flex items-center justify-between">
   381→          <div className="flex items-center gap-2">
   382→            {currentJob.status === 'running' || currentJob.status === 'pending' ? (
   383→              <div className="relative">
   384→                <Loader2 className="h-4 w-4 animate-spin text-primary" />
   385→                <div className="absolute inset-0 rounded-full animate-ping bg-primary/20" />
   386→              </div>
   387→            ) : currentJob.status === 'completed' ? (
   388→              <CheckCircle2 className="h-4 w-4 text-emerald-500" />
   389→            ) : (
   390→              <XCircle className="h-4 w-4 text-red-500" />
   391→            )}
   392→            <span className="text-sm font-medium">
   393→              {currentJob.status === 'pending' ? 'Queuing discovery...' :
   394→                currentJob.status === 'running' ? 'Discovering leads...' :
   395→                currentJob.status === 'completed' ? 'Discovery complete!' :
   396→                'Discovery failed'}
   397→            </span>
   398→          </div>
   399→          <Badge variant="outline" className={cn('text-[10px] border', STATUS_STYLES[currentJob.status])}>
   400→            {currentJob.status}
   401→          </Badge>
   402→        </div>
   403→        <Progress value={progress} className="h-2" />
   404→        {/* Progress steps */}
   405→        <div className="flex items-center gap-1">
   406→          {progressSteps.map((step, i) => (
   407→            <div key={step.label} className="flex items-center gap-1 flex-1">
   408→              <div className={cn(
   409→                'h-1.5 w-1.5 rounded-full shrink-0 transition-colors',
   410→                step.done ? 'bg-primary' : 'bg-muted-foreground/20'
   411→              )} />
   412→              <span className={cn(
   413→                'text-[9px] truncate transition-colors',
   414→                step.done ? 'text-primary font-medium' : 'text-muted-foreground/50'
   415→              )}>{step.label}</span>
   416→              {i < progressSteps.length - 1 && (
   417→                <div className={cn('flex-1 h-px', step.done ? 'bg-primary/40' : 'bg-muted-foreground/10')} />
   418→              )}
   419→            </div>
   420→          ))}
   421→        </div>
   422→        <div className="flex items-center justify-between text-xs text-muted-foreground">
   423→          <span>{currentJob.niche} · {currentJob.country}{currentJob.city ? ` · ${currentJob.city}` : ''}</span>
   424→          {currentJob.status === 'completed' && (
   425→            <span className="text-emerald-500 font-medium">
   426→              {currentJob.leadsAdded} verified lead{currentJob.leadsAdded === 1 ? '' : 's'} found
   427→              {currentJob.filteredOut ? ` · ${currentJob.filteredOut} rejected by hard filters` : ''}
   428→            </span>
   429→          )}
   430→          {currentJob.status === 'failed' && (currentJob.errorMessage || currentJob.message) && (
   431→            <span className="text-red-500 truncate max-w-[280px]" title={currentJob.errorMessage || currentJob.message}>
   432→              {currentJob.errorMessage || currentJob.message}
   433→            </span>
   434→          )}
   435→        </div>
   436→      </CardContent>
   437→    </Card>
   438→  );
   439→}
